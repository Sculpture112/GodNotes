#include <bits/stdc++.h>
using namespace std;

using ll = long long;

const int INF = 0x3f3f3f3f;
const ll LINF = 4e18;

#define all(x) (x).begin(), (x).end()

void solve() {
    string a, b;
    cin >> a >> b;

    // 去除前导零
    size_t posA = a.find_first_not_of('0');
    size_t posB = b.find_first_not_of('0');

    if (posA == string::npos || posB == string::npos) {
        cout << 0 << '\n';
        return;
    }

    a = a.substr(posA);
    b = b.substr(posB);

    vector<int> result(a.size() + b.size(), 0);

    // 模拟竖式乘法
    for (int i = static_cast<int>(a.size()) - 1; i >= 0; --i) {
        for (int j = static_cast<int>(b.size()) - 1; j >= 0; --j) {
            result[i + j + 1] += (a[i] - '0') * (b[j] - '0');
        }
    }

    // 统一进位
    for (int i = static_cast<int>(result.size()) - 1; i > 0; --i) {
        result[i - 1] += result[i] / 10;
        result[i] %= 10;
    }

    // 输出，跳过最高位前导零
    int start = 0;
    while (start < static_cast<int>(result.size()) - 1 &&
           result[start] == 0) {
        ++start;
    }

    for (int i = start; i < static_cast<int>(result.size()); ++i) {
        cout << result[i];
    }
    cout << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();

    return 0;
}
