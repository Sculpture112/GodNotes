```cpp
#include <bits/stdc++.h>
using namespace std;
// 最短无序连续子数组
// 给你一个整数数组nums，你需要找出一个 连续子数组
// 如果对这个子数组进行升序排序，那么整个数组都会变为升序排序
// 请你找出符合题意的最短子数组，并输出它的长度
// 测试链接 : https://leetcode.cn/problems/shortest-unsorted-continuous-subarray/
class Solution { public: int findUnsortedSubarray(vector<int>& nums) { int n=nums.size(), right=-1, mx=INT_MIN, left=n, mn=INT_MAX; for(int i=0;i<n;i++){if(mx>nums[i])right=i;mx=max(mx,nums[i]);} for(int i=n-1;i>=0;i--){if(mn<nums[i])left=i;mn=min(mn,nums[i]);} return max(0,right-left+1); } };
```
