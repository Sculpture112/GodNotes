#include <bits/stdc++.h>

using namespace std;

// 用递归函数逆序栈

// 栈底元素移除掉，上面的元素盖下来
// 返回移除掉的栈底元素
int bottomOut(stack<int>& stack) {
    int ans = stack.top();
    stack.pop();
    if (stack.empty()) {
        return ans;
    } else {
        int last = bottomOut(stack);
        stack.push(ans);
        return last;
    }
}

void reverse(stack<int>& stack) {
    if (stack.empty()) {
        return;
    }
    int num = bottomOut(stack);
    reverse(stack);
    stack.push(num);
}
