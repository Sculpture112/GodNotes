```cpp
#include <bits/stdc++.h>

using namespace std;

// 利用先序与中序遍历序列构造二叉树
// 测试链接 : https://leetcode.cn/problems/construct-binary-tree-from-preorder-and-inorder-traversal/
class Solution {
public:
    unordered_map<int, int> pos;

    TreeNode* buildTree(vector<int>& pre, vector<int>& in) {
        if (pre.size() != in.size()) return nullptr;
        for (int i = 0; i < (int)in.size(); i++) pos[in[i]] = i;
        return f(pre, 0, pre.size() - 1, in, 0, in.size() - 1);
    }

    TreeNode* f(vector<int>& pre, int l1, int r1, vector<int>& in, int l2, int r2) {
        if (l1 > r1) return nullptr;
        TreeNode* head = new TreeNode(pre[l1]);
        if (l1 == r1) return head;
        int k = pos[pre[l1]];
        // pre : l1(........)[.......r1]
        // in  : (l2......)k[........r2]
        // (...)是左树对应，[...]是右树的对应
        head->left = f(pre, l1 + 1, l1 + k - l2, in, l2, k - 1);
        head->right = f(pre, l1 + k - l2 + 1, r1, in, k + 1, r2);
        return head;
    }
};
```
