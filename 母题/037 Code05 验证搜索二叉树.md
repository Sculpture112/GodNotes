#include <bits/stdc++.h>

using namespace std;

// 验证搜索二叉树
// 测试链接 : https://leetcode.cn/problems/validate-binary-search-tree/
class Solution {
public:
    // 提交时改名为isValidBST
    bool isValidBST1(TreeNode* head) {
        if (head == nullptr) return true;
        TreeNode* pre = nullptr;
        stack<TreeNode*> st;
        while (!st.empty() || head != nullptr) {
            if (head != nullptr) {
                st.push(head);
                head = head->left;
            } else {
                head = st.top();
                st.pop();
                if (pre != nullptr && pre->val >= head->val) return false;
                pre = head;
                head = head->right;
            }
        }
        return true;
    }

    long long minValue, maxValue;

    // 提交时改名为isValidBST
    bool isValidBST(TreeNode* head) {
        if (head == nullptr) {
            minValue = LLONG_MAX;
            maxValue = LLONG_MIN;
            return true;
        }
        bool lok = isValidBST(head->left);
        long long lmin = minValue;
        long long lmax = maxValue;
        bool rok = isValidBST(head->right);
        long long rmin = minValue;
        long long rmax = maxValue;
        minValue = min(min(lmin, rmin), (long long)head->val);
        maxValue = max(max(lmax, rmax), (long long)head->val);
        return lok && rok && lmax < head->val && head->val < rmin;
    }
};
