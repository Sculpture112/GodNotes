```cpp
#include <bits/stdc++.h>
using namespace std;
// 超级洗衣机
// 假设有n台超级洗衣机放在同一排上
// 开始的时候，每台洗衣机内可能有一定量的衣服，也可能是空的
// 在每一步操作中，你可以选择任意 m (1 <= m <= n) 台洗衣机，同时将每台洗衣机的一件衣服送到相邻的一台洗衣机
// 给定一个整数数组machines代表从左至右每台洗衣机中的衣物数量
// 请给出能让所有洗衣机中剩下的衣物的数量相等的最少的操作步数
// 如果不能使每台洗衣机中衣物的数量相等则返回-1
// 测试链接 : https://leetcode.cn/problems/super-washing-machines/
class Solution { public: int findMinMoves(vector<int>& arr) { int n = arr.size(), sum = accumulate(arr.begin(), arr.end(), 0); if (sum % n) return -1; int avg = sum / n, leftSum = 0, ans = 0; for (int i = 0; i < n; leftSum += arr[i], i++) { int leftNeed = i * avg - leftSum; int rightNeed = (n - i - 1) * avg - (sum - leftSum - arr[i]); int bottleNeck = leftNeed > 0 && rightNeed > 0 ? leftNeed + rightNeed : max(abs(leftNeed), abs(rightNeed)); ans = max(ans, bottleNeck); } return ans; } };
```
