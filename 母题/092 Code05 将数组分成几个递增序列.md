#include <bits/stdc++.h>
using namespace std;
class Solution {
public:
    bool canDivideIntoSubsequences(vector<int>& nums, int k) {
        int cnt = 1, maxCnt = 1;
        for (int i = 1; i < nums.size(); i++) {
            if (nums[i] == nums[i - 1]) cnt++;
            else { maxCnt = max(maxCnt, cnt); cnt = 1; }
        }
        return nums.size() / max(maxCnt, cnt) >= k;
    }
};
