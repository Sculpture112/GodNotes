```cpp
#include <bits/stdc++.h>
using namespace std;
class Solution{public:bool canDivideIntoSubsequences(vector<int>&a,int k){int c=1,m=1;for(int i=1;i<a.size();i++)a[i]==a[i-1]?c++:(m=max(m,c),c=1);return a.size()/max(m,c)>=k;}};
```
