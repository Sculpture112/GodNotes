#include <bits/stdc++.h>
using namespace std;
// 执行所有任务的最少初始电量
// 每一个任务有两个参数，需要耗费的电量、至少多少电量才能开始这个任务
// 返回手机至少需要多少的初始电量，才能执行完所有的任务
// 测试链接 : https://leetcode.cn/problems/minimum-initial-energy-to-finish-tasks/
class Solution {
public:
    int minimumEffort(vector<vector<int>>& tasks) { return atLeast2(tasks); }
    int f(vector<vector<int>>& jobs, int i) {
        if (i == jobs.size()) { int ans = 0; for (auto& job : jobs) ans = max(job[1], ans + job[0]); return ans; }
        int ans = INT_MAX;
        for (int j = i; j < jobs.size(); j++) { swap(jobs[i], jobs[j]); ans = min(ans, f(jobs, i + 1)); swap(jobs[i], jobs[j]); }
        return ans;
    }
    int atLeast1(vector<vector<int>>& jobs) { return f(jobs, 0); }
    int atLeast2(vector<vector<int>>& jobs) {
        sort(jobs.begin(), jobs.end(), [](const auto& a, const auto& b) { return a[0] - a[1] > b[0] - b[1]; });
        int ans = 0;
        for (auto& job : jobs) ans = max(ans + job[0], job[1]);
        return ans;
    }
};
