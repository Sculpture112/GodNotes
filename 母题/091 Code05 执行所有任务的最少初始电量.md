// 执行所有任务的最少初始电量
// 每一个任务有两个参数，需要耗费的电量、至少多少电量才能开始这个任务
// 返回手机至少需要多少的初始电量，才能执行完所有的任务
// 测试链接 : https://leetcode.cn/problems/minimum-initial-energy-to-finish-tasks/
class Solution{public:int minimumEffort(vector<vector<int>>&t){return atLeast2(t);}int f(vector<vector<int>>&a,int i){if(i==a.size()){int ans=0;for(auto&x:a)ans=max(x[1],ans+x[0]);return ans;}int ans=INT_MAX;for(int j=i;j<a.size();j++){swap(a[i],a[j]);ans=min(ans,f(a,i+1));swap(a[i],a[j]);}return ans;}int atLeast1(vector<vector<int>>&a){return f(a,0);}int atLeast2(vector<vector<int>>&a){sort(a.begin(),a.end(),[](auto&x,auto&y){return x[0]-x[1]>y[0]-y[1];});int ans=0;for(auto&x:a)ans=max(ans+x[0],x[1]);return ans;}};
