```cpp
#include <bits/stdc++.h>
using namespace std;
// 执行所有任务的最少初始电量
// 每一个任务有两个参数，需要耗费的电量、至少多少电量才能开始这个任务
// 返回手机至少需要多少的初始电量，才能执行完所有的任务
// 测试链接 : https://leetcode.cn/problems/minimum-initial-energy-to-finish-tasks/
class Solution{public:int minimumEffort(vector<vector<int>>&tasks){sort(tasks.begin(),tasks.end(),[](auto&a,auto&b){return a[0]-a[1]>b[0]-b[1];});int ans=0;for(auto&x:tasks)ans=max(ans+x[0],x[1]);return ans;}};
```
