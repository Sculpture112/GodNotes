#include <bits/stdc++.h>

using namespace std;

// 丁香之路，java版
// 一共有n个点，点i和点j的距离为|i - j|，所以这是完全图，边是无向边
// 给定m个点对，每个点对(a, b)，保证 a != b，表示a和b之间的无向边做了标记
// 给定起点s，不管终点是什么，每条被标记的无向边，走过至少一次
// 打印起点s，终点i的情况下，走过路程的最小距离是多少
// 终点i可以是1~n中的任意一点，所以一共打印n个数值
// 1 <= n <= 2500
// m <= n个顶点的完全图的边数
// 测试链接 : https://www.luogu.com.cn/problem/P6628
// 提交以下的code，提交时请把类名改成"Main"，可以通过所有测试用例

const int MAXN = 3001;
const int MAXM = 3000001;
int n, m, s;
int a[MAXM];
int b[MAXM];

bool fixedNode[MAXN];
set<int> nodeSet;
int nodeArray[MAXN];
int nodeCount;

int father1[MAXN];
int degree1[MAXN];
int father2[MAXN];
int degree2[MAXN];

struct Edge {
    int u;
    int v;
    int w;
};

Edge edgeArray[MAXN];
long long sum;

bool edgeCompare(const Edge& edge1, const Edge& edge2) {
    if (edge1.w != edge2.w) {
        return edge1.w < edge2.w;
    }
    if (edge1.u != edge2.u) {
        return edge1.u < edge2.u;
    }
    return edge1.v < edge2.v;
}

int distanceBetween(int x, int y) {
    return abs(x - y);
}

int findFather(int father[], int x) {
    if (x != father[x]) {
        father[x] = findFather(father, father[x]);
    }
    return father[x];
}

void mergeSets(int father[], int x, int y) {
    int fx = findFather(father, x);
    int fy = findFather(father, y);
    if (fx != fy) {
        father[fx] = fy;
    }
}

long long kruskal() {
    long long cost = 0;
    int edgeCount = nodeCount - 1;
    sort(edgeArray + 1, edgeArray + edgeCount + 1, edgeCompare);
    for (int i = 1; i <= edgeCount; i++) {
        int u = edgeArray[i].u;
        int v = edgeArray[i].v;
        int w = edgeArray[i].w;
        if (findFather(father2, u) != findFather(father2, v)) {
            cost += 1LL * w * 2;
            mergeSets(father2, u, v);
        }
    }
    return cost;
}

long long compute(int start, int end) {
    nodeCount = 0;
    for (int x : nodeSet) {
        nodeArray[++nodeCount] = x;
    }
    for (int i = 1; i <= n; i++) {
        father2[i] = findFather(father1, i);
        degree2[i] = degree1[i];
    }
    long long answer = sum;
    degree2[start]++;
    degree2[end]++;
    mergeSets(father2, start, end);
    for (int i = 1, p = 0; i <= nodeCount; i++) {
        if ((degree2[nodeArray[i]] & 1) == 1) {
            if (p == 0) {
                p = i;
            } else {
                answer += distanceBetween(nodeArray[p], nodeArray[i]);
                while (p < i) {
                    mergeSets(father2, nodeArray[p++], nodeArray[i]);
                }
                p = 0;
            }
        }
    }
    for (int i = 1; i <= nodeCount - 1; i++) {
        edgeArray[i].u = nodeArray[i];
        edgeArray[i].v = nodeArray[i + 1];
        edgeArray[i].w = nodeArray[i + 1] - nodeArray[i];
    }
    answer += kruskal();
    return answer;
}

void prepare() {
    fixedNode[s] = true;
    nodeSet.insert(s);
    sum = 0;
    for (int i = 1; i <= n; i++) {
        father1[i] = i;
    }
    for (int i = 1, u, v; i <= m; i++) {
        u = a[i];
        v = b[i];
        degree1[u]++;
        degree1[v]++;
        fixedNode[u] = true;
        fixedNode[v] = true;
        nodeSet.insert(u);
        nodeSet.insert(v);
        mergeSets(father1, u, v);
        sum += distanceBetween(u, v);
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m >> s;
    for (int i = 1; i <= m; i++) {
        cin >> a[i] >> b[i];
    }
    prepare();
    for (int i = 1; i <= n; i++) {
        if (!fixedNode[i]) {
            nodeSet.insert(i);
        }
        cout << compute(s, i) << " ";
        if (!fixedNode[i]) {
            nodeSet.erase(i);
        }
    }
    cout << '\n';
    return 0;
}
