#include <bits/stdc++.h>

using namespace std;

// 邮递员，C++版
// 一共n个点，m条有向边，不会有重边和自环
// 邮递员一定要从1号点开始，最后回到1号点，每条边都要走1次
// 邮递员依次走过的点的编号，就可以形成一个路径
// 给定t个序列，每个序列先给定长度k，然后是k个节点的编号
// 希望每个序列都是路径的连续子段，如果不存在这样的路径，打印"NIE"
// 如果存在这样的路径，先打印"TAK"，然后打印依次走过的节点编号
// 2 <= n <= 5 * 10^4    1 <= m <= 2 * 10^5
// 0 <= t <= 10^4        所有序列的节点总量 <= 2 * 10^6
// 测试链接 : https://www.luogu.com.cn/problem/P3443
// 如下实现是C++的版本，C++版本和java版本逻辑完全一样
// 提交如下代码，可以通过所有测试用例

const int MAXN = 50005;
const int MAXM = 200005;
const int MAXC = 2000005;
int n, m, t;
int a[MAXM];
int b[MAXM];
int sequenceData[MAXC];

map<pair<int, int>, int> pairEdge;

int graphHead[MAXN];
int nextGraphEdge[MAXM];
int graphTo[MAXM];
int chainHead[MAXM];
int graphEdgeCount;
int compressedEdgeCount;

int currentEdge[MAXN];
int inDegree[MAXN];
int outDegree[MAXN];

bool isHead[MAXM];
int edgeToEdge[MAXM];
bool visited[MAXM];

int path[MAXM];
int pathCount;

int answer[MAXM];
int answerCount;

void addEdge(int x, int y, int chain) {
    nextGraphEdge[++graphEdgeCount] = graphHead[x];
    graphTo[graphEdgeCount] = y;
    chainHead[graphEdgeCount] = chain;
    graphHead[x] = graphEdgeCount;
}

bool linkEdge() {
    for (int i = 1; i <= m; i++) {
        pairEdge[{a[i], b[i]}] = i;
        isHead[i] = true;
    }
    int size = sequenceData[1];
    int left = 2;
    int right = left + size - 1;
    int x, y, leftEdge, rightEdge;
    while (size > 0) {
        leftEdge = 0;
        for (int i = left; i < right; i++) {
            x = sequenceData[i];
            y = sequenceData[i + 1];
            auto it = pairEdge.find({x, y});
            if (it == pairEdge.end()) {
                return false;
            }
            rightEdge = it->second;
            if (leftEdge != 0) {
                if (edgeToEdge[leftEdge] != 0 && edgeToEdge[leftEdge] != rightEdge) {
                    return false;
                }
                edgeToEdge[leftEdge] = rightEdge;
                isHead[rightEdge] = false;
            }
            leftEdge = rightEdge;
        }
        size = sequenceData[right + 1];
        left = right + 2;
        right = left + size - 1;
    }
    return true;
}

int getLinkEnd(int edge) {
    while (edgeToEdge[edge] != 0) {
        if (visited[edge]) {
            return -1;
        }
        visited[edge] = true;
        edge = edgeToEdge[edge];
    }
    return edge;
}

bool compress() {
    for (int i = 1; i <= m; i++) {
        if (isHead[i]) {
            int x = a[i];
            int y = b[i];
            if (edgeToEdge[i] != 0) {
                int end = getLinkEnd(i);
                if (end == -1) {
                    return false;
                }
                y = b[end];
            }
            outDegree[x]++;
            inDegree[y]++;
            compressedEdgeCount++;
            addEdge(x, y, i);
        }
    }
    for (int i = 1; i <= n; i++) {
        if (inDegree[i] != outDegree[i]) {
            return false;
        }
    }
    for (int i = 1; i <= n; i++) {
        currentEdge[i] = graphHead[i];
    }
    return true;
}

void euler(int u, int chain) {
    for (int edge = currentEdge[u]; edge > 0; edge = currentEdge[u]) {
        currentEdge[u] = nextGraphEdge[edge];
        euler(graphTo[edge], chainHead[edge]);
    }
    path[++pathCount] = chain;
}

void decompress() {
    answer[++answerCount] = 1;
    for (int i = pathCount - 1; i >= 1; i--) {
        int edge = path[i];
        while (edge > 0) {
            answer[++answerCount] = b[edge];
            edge = edgeToEdge[edge];
        }
    }
}

bool compute() {
    if (!linkEdge()) {
        return false;
    }
    if (!compress()) {
        return false;
    }
    euler(1, -1);
    if (pathCount != compressedEdgeCount + 1) {
        return false;
    }
    decompress();
    return answerCount == m + 1;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m;
    for (int i = 1; i <= m; i++) {
        cin >> a[i] >> b[i];
    }
    cin >> t;
    int index = 0;
    for (int i = 1, size; i <= t; i++) {
        cin >> size;
        sequenceData[++index] = size;
        for (int j = 1; j <= size; j++) {
            cin >> sequenceData[++index];
        }
    }
    bool check = compute();
    if (!check) {
        cout << "NIE\n";
    } else {
        cout << "TAK\n";
        for (int i = 1; i <= answerCount; i++) {
            cout << answer[i] << "\n";
        }
    }
    return 0;
}
