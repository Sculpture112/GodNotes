```cpp
#include <bits/stdc++.h>
using namespace std;
// 数组的最小偏移量
// 测试链接 : https://leetcode.cn/problems/minimize-deviation-in-array/
class Solution{public:int minimumDeviation(vector<int>&a){multiset<int>s;for(int x:a)s.insert(x&1?x*2:x);int ans=*s.rbegin()-*s.begin();while(ans&&!*s.rbegin()%2){int x=*s.rbegin();s.erase(prev(s.end()));s.insert(x/2);ans=min(ans,*s.rbegin()-*s.begin());}return ans;}};
```
