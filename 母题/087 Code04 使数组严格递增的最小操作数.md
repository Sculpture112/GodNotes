```cpp
#include <bits/stdc++.h>
using namespace std;
// 使数组严格递增的最小操作数
// 给你两个整数数组 arr1 和 arr2
// 返回使 arr1 严格递增所需要的最小操作数（可能为0）
// 每一步操作中，你可以分别从 arr1 和 arr2 中各选出一个索引
// 分别为 i 和 j，0 <= i < arr1.length 和 0 <= j < arr2.length
// 然后进行赋值运算 arr1[i] = arr2[j]
// 如果无法让 arr1 严格递增，请返回-1
// 1 <= arr1.length, arr2.length <= 2000
// 0 <= arr1[i], arr2[i] <= 10^9
// 测试链接 : https://leetcode.cn/problems/make-array-strictly-increasing/
class Solution { public: int makeArrayIncreasing(vector<int>& arr1, vector<int>& arr2) { sort(arr2.begin(),arr2.end()); arr2.erase(unique(arr2.begin(),arr2.end()),arr2.end()); map<int,int> dp; dp[INT_MIN]=0; for(int x:arr1){map<int,int> next;for(auto [pre,c]:dp){if(x>pre) next[x]=min(next.count(x)?next[x]:INT_MAX,c);auto it=upper_bound(arr2.begin(),arr2.end(),pre);if(it!=arr2.end()) next[*it]=min(next.count(*it)?next[*it]:INT_MAX,c+1);}dp.swap(next);}int ans=INT_MAX;for(auto [x,c]:dp)ans=min(ans,c);return ans==INT_MAX?-1:ans;} };
```
