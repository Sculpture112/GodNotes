#include <bits/stdc++.h>
using namespace std;
// 分成k份的最大乘积
// 一个数字n一定要分成k份，得到的乘积尽量大是多少
// 数字n和k，可能非常大，到达10^12规模
// 结果可能更大，所以返回结果对 1000000007 取模
// 来自真实大厂笔试，没有在线测试，对数器验证
long long power(long long x,long long n,int mod){long long ans=1;while(n){if(n&1)ans=ans*x%mod;x=x*x%mod;n>>=1;}return ans;}
int maxValue2(long long n,long long k){const int mod=1000000007;long long a=n/k,b=n%k;return power(a+1,b,mod)*power(a,k-b,mod)%mod;}
