#include <bits/stdc++.h>
using namespace std;
// 组团买票找到了在线测试

// 实现做了一些小优化，具体可以看注释
// 测试链接 : https://www.luogu.com.cn/problem/P12331
// 提交以下的code，提交时请把类名改成"Main"，可以通过所有测试用例
struct Game {
    long long ki, bi; int people = 0;
    Game(long long k, long long b) : ki(k), bi(b) {}
    long long earn() const { return cost(people + 1) - cost(people); }
    long long cost(long long p) const { return p * max(ki * p + bi, 0LL); }
};
struct Compare { bool operator()(const Game& a, const Game& b) const { return a.earn() < b.earn(); } };
int main() {
    ios::sync_with_stdio(false); cin.tie(nullptr);
    int n, m; cin >> n >> m;
    priority_queue<Game, vector<Game>, Compare> heap;
    for (int i = 0; i < m; i++) { long long k, b; cin >> k >> b; Game cur(k, b); if (cur.earn() > 0) heap.push(cur); }
    long long ans = 0;
    for (int i = 0; i < n && !heap.empty(); i++) { Game cur = heap.top(); heap.pop(); long long money = cur.earn(); if (money <= 0) break; ans += money; cur.people++; if (cur.earn() > 0) heap.push(cur); }
    cout << ans << '\n';
    return 0;
}
