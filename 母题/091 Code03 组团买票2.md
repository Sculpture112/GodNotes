struct Game{long long k,b;int p=0;long long cost(long long x)const{return x*max(k*x+b,0LL);}long long earn()const{return cost(p+1)-cost(p);}};struct Cmp{bool operator()(const Game&a,const Game&b)const{return a.earn()<b.earn();}};
int main(){ios::sync_with_stdio(false);cin.tie(nullptr);int n,m;cin>>n>>m;priority_queue<Game,vector<Game>,Cmp>h;for(int i=0;i<m;i++){Game x;cin>>x.k>>x.b;if(x.earn()>0)h.push(x);}long long ans=0;while(n--&&!h.empty()){auto x=h.top();h.pop();if(x.earn()<=0)break;ans+=x.earn();x.p++;if(x.earn()>0)h.push(x);}cout<<ans<<'\n';}
