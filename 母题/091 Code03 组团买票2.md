#include <bits/stdc++.h>
using namespace std;
// 组团买票找到了在线测试
// 逻辑和课上讲的一样，但是测试中设定的ki为负数
// 测试链接 : https://www.luogu.com.cn/problem/P12331
struct Game {
    long long ki, bi;
    int people = 0;
    long long cost(long long p) const { return p * max(ki * p + bi, 0LL); }
    long long earn() const { return cost(people + 1) - cost(people); }
};

struct Compare {
    bool operator()(const Game& a, const Game& b) const { return a.earn() < b.earn(); }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    priority_queue<Game, vector<Game>, Compare> heap;
    for (int i = 0; i < m; i++) {
        Game cur;
        cin >> cur.ki >> cur.bi;
        if (cur.earn() > 0) heap.push(cur);
    }
    long long ans = 0;
    for (int i = 0; i < n && !heap.empty(); i++) {
        Game cur = heap.top();
        heap.pop();
        long long money = cur.earn();
        if (money <= 0) break;
        ans += money;
        cur.people++;
        if (cur.earn() > 0) heap.push(cur);
    }
    cout << ans << '\n';
    return 0;
}
