```cpp
#include <bits/stdc++.h>
using namespace std;

// 自由之路
// 题目描述比较多，打开链接查看
// 测试链接 : https://leetcode.cn/problems/freedom-trail/
class Solution {
public:
    int n, m;
    string ring, key;
    vector<vector<int>> where;
    vector<vector<int>> dp;

    int findRotateSteps(string r, string k) {
        ring = r;
        key = k;
        n = ring.size();
        m = key.size();
        where.assign(26, {});
        for (int i = 0; i < n; i++) {
            where[ring[i] - 'a'].push_back(i);
        }
        dp.assign(n, vector<int>(m, -1));
        return f(0, 0);
    }

    int f(int i, int j) {
        if (j == m) {
            return 0;
        }
        if (dp[i][j] != -1) {
            return dp[i][j];
        }
        if (ring[i] == key[j]) {
            return dp[i][j] = 1 + f(i, j + 1);
        }

        int v = key[j] - 'a';
        int jump1 = clock(i, v);
        int distance1 = jump1 > i ? jump1 - i : n - i + jump1;
        int jump2 = counterClock(i, v);
        int distance2 = i > jump2 ? i - jump2 : i + n - jump2;
        return dp[i][j] = min(distance1 + f(jump1, j), distance2 + f(jump2, j));
    }

    int clock(int i, int v) {
        auto& arr = where[v];
        auto it = upper_bound(arr.begin(), arr.end(), i);
        return it == arr.end() ? arr[0] : *it;
    }

    int counterClock(int i, int v) {
        auto& arr = where[v];
        auto it = lower_bound(arr.begin(), arr.end(), i);
        return it == arr.begin() ? arr.back() : *(--it);
    }
};
```
