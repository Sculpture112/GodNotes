#include <bits/stdc++.h>
using namespace std;
class Solution {
public:
    int minRefuelStops(int target, int startFuel, vector<vector<int>>& stations) {
        priority_queue<int> heap;
        long long to = startFuel;
        int ans = 0;
        for (auto& station : stations) {
            while (to < station[0] && !heap.empty()) { to += heap.top(); heap.pop(); ans++; }
            if (to < station[0]) return -1;
            heap.push(station[1]);
        }
        while (to < target && !heap.empty()) { to += heap.top(); heap.pop(); ans++; }
        return to >= target ? ans : -1;
    }
};
