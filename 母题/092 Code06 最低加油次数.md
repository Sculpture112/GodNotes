```cpp
#include <bits/stdc++.h>
using namespace std;
// 最低加油次数
// 汽车从起点出发驶向目的地，该目的地位于出发位置东面target英里处
// 沿途有加油站，用数组stations表示，其中 stations[i] = [positioni, fueli]
// 假设汽车油箱的容量是无限的，其中最初有startFuel升燃料
// 它每行驶1英里就会用掉1升汽油
// 为了到达目的地，汽车所必要的最低加油次数是多少？如果无法到达，则返回-1
// 测试链接 : https://leetcode.cn/problems/minimum-number-of-refueling-stops/
class Solution { public: int minRefuelStops(int target, int startFuel, vector<vector<int>>& stations) { if (startFuel >= target) return 0; priority_queue<int> heap; long long to = startFuel; int cnt = 0; for (auto& station : stations) { int position = station[0], fuel = station[1]; if (to < position) { while (!heap.empty() && to < position) { to += heap.top(); heap.pop(); cnt++; if (to >= target) return cnt; } if (to < position) return -1; } heap.push(fuel); } while (!heap.empty()) { to += heap.top(); heap.pop(); if (++cnt && to >= target) return cnt; } return -1; } };
```
