```cpp
#include <bits/stdc++.h>
using namespace std;
class Solution{public:int minRefuelStops(int target,int fuel,vector<vector<int>>&s){priority_queue<int>h;long long to=fuel;int ans=0;for(auto&x:s){while(to<x[0]&&!h.empty())to+=h.top(),h.pop(),ans++;if(to<x[0])return -1;h.push(x[1]);}while(to<target&&!h.empty())to+=h.top(),h.pop(),ans++;return to>=target?ans:-1;}};
```
