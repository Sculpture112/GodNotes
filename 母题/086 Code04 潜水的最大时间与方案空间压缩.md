#include <bits/stdc++.h>

using namespace std;

// 潜水的最大时间与方案
// 一共有n个工具，每个工具都有自己的重量a、阻力b、提升的停留时间c
// 因为背包有限，所以只能背重量不超过m的工具
// 因为力气有限，所以只能背阻力不超过v的工具
// 希望能在水下停留的时间最久
// 返回最久的停留时间和下标字典序最小的选择工具的方案
// 注意这道题的字典序设定（根据提交的结果推论的）：
// 下标方案整体构成的字符串保证字典序最小
// 比如下标方案"1 120"比下标方案"1 2"字典序小
// 测试链接 : https://www.luogu.com.cn/problem/P1759
// 请同学们务必参考如下代码中关于输入、输出的处理
// 这是输入输出处理效率很高的写法
// 提交以下的code，提交时请把类名改成"Main"，可以直接通过

// 本文件做了空间压缩优化
// 可以通过全部测试用例

int m, v, n;
vector<int> a, b, c;
vector<vector<int>> dp;
vector<vector<string>> path;

// 多维费用背包的空间压缩版本
// 请务必掌握空间压缩技巧
// 之前的课讲了很多遍了
void compute() {
    string p2;
    for (int i = 1; i <= n; i++) {
        for (int j = m; j >= a[i]; j--) {
            for (int k = v; k >= b[i]; k--) {
                if (path[j - a[i]][k - b[i]].empty()) p2 = to_string(i);
                else p2 = path[j - a[i]][k - b[i]] + " " + to_string(i);
                int value = dp[j - a[i]][k - b[i]] + c[i];
                if (dp[j][k] < value) {
                    dp[j][k] = value;
                    path[j][k] = p2;
                } else if (dp[j][k] == value && p2 < path[j][k]) {
                    path[j][k] = p2;
                }
            }
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    while (cin >> m >> v >> n) {
        a.assign(n + 1, 0);
        b.assign(n + 1, 0);
        c.assign(n + 1, 0);
        for (int i = 1; i <= n; i++) cin >> a[i] >> b[i] >> c[i];
        dp.assign(m + 1, vector<int>(v + 1, 0));
        path.assign(m + 1, vector<string>(v + 1));
        compute();
        cout << dp[m][v] << '\n';
        cout << path[m][v] << '\n';
    }
    return 0;
}
