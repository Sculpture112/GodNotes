#include <bits/stdc++.h>

using namespace std;

// 有向图的欧拉路径，java版
// 图中有n个点，m条有向边，每条边给出两个端点
// 如果存在欧拉路径，输出字典序最小的结果，如果不存在打印No
// 1 <= n <= 10^5
// 1 <= m <= 2 * 10^5
// 测试链接 : https://www.luogu.com.cn/problem/P7771
// 提交以下的code，提交时请把类名改成"Main"，可以通过所有测试用例

const int MAXN = 100001;
const int MAXM = 200002;

struct Edge {
    int u;
    int v;
};

int n, m;
Edge edgeArr[MAXM];
int head[MAXN];
int nxt[MAXM];
int to[MAXM];
int cntg;
int cur[MAXN];
int outDeg[MAXN];
int inDeg[MAXN];
int path[MAXM];
int cntp;
int sta[MAXM];
int top;

void addEdge(int u, int v) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    head[u] = cntg;
}

void connectGraph() {
    sort(edgeArr + 1, edgeArr + m + 1, [](const Edge& e1, const Edge& e2) {
        if (e1.u != e2.u) {
            return e1.u < e2.u;
        }
        return e1.v < e2.v;
    });
    for (int l = 1, r = 1; l <= m; l = ++r) {
        while (r + 1 <= m && edgeArr[l].u == edgeArr[r + 1].u) {
            r++;
        }
        for (int i = r; i >= l; i--) {
            int u = edgeArr[i].u;
            int v = edgeArr[i].v;
            outDeg[u]++;
            inDeg[v]++;
            addEdge(u, v);
        }
    }
    for (int i = 1; i <= n; i++) {
        cur[i] = head[i];
    }
}

// 有向图中找到一个起点，去生成欧拉回路 或者 欧拉路径
int directedStart() {
    int start = -1;
    int finish = -1;
    for (int i = 1; i <= n; i++) {
        int value = outDeg[i] - inDeg[i];
        if (value < -1 || value > 1 || (value == 1 && start != -1) || (value == -1 && finish != -1)) {
            return -1;
        }
        if (value == 1) {
            start = i;
        }
        if (value == -1) {
            finish = i;
        }
    }
    if ((start == -1) ^ (finish == -1)) {
        return -1;
    }
    if (start != -1) {
        return start;
    }
    for (int i = 1; i <= n; i++) {
        if (outDeg[i] > 0) {
            return i;
        }
    }
    return -1;
}

// Hierholzer算法递归版，java会爆栈，C++可以通过
void euler1(int u) {
    for (int e = cur[u]; e > 0; e = cur[u]) {
        cur[u] = nxt[e];
        euler1(to[e]);
    }
    path[++cntp] = u;
}

// Hierholzer算法迭代版
void euler2(int node) {
    top = 0;
    sta[++top] = node;
    while (top > 0) {
        int u = sta[top--];
        int e = cur[u];
        if (e != 0) {
            cur[u] = nxt[e];
            sta[++top] = u;
            sta[++top] = to[e];
        } else {
            path[++cntp] = u;
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m;
    for (int i = 1; i <= m; i++) {
        cin >> edgeArr[i].u >> edgeArr[i].v;
    }
    connectGraph();
    int start = directedStart();
    if (start == -1) {
        cout << "No\n";
    } else {
        // euler1(start);
        euler2(start);
        if (cntp != m + 1) {
            cout << "No\n";
        } else {
            for (int i = cntp; i >= 1; i--) {
                cout << path[i] << " ";
            }
            cout << '\n';
        }
    }
    return 0;
}
