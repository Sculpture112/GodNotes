```cpp
#include <bits/stdc++.h>
using namespace std;

// 每个人戴不同帽子的方案数
// 总共有 n 个人和 40 种不同的帽子，帽子编号从 1 到 40
// 给你一个整数列表的列表 hats ，其中 hats[i] 是第 i 个人所有喜欢帽子的列表
// 请你给每个人安排一顶他喜欢的帽子，确保每个人戴的帽子跟别人都不一样，并返回方案数
// 由于答案可能很大，答案对 1000000007 取模
// 测试链接 : https://leetcode.cn/problems/number-of-ways-to-wear-different-hats-to-each-other
class Solution {
public:
    int MOD = 1000000007;

    int numberWays(vector<vector<int>>& arr) {
        int m = 0;
        for (auto& person : arr) {
            for (int hat : person) {
                m = max(m, hat);
            }
        }

        int n = arr.size();
        vector<int> hats(m + 1, 0);
        for (int pi = 0; pi < n; pi++) {
            for (int hat : arr[pi]) {
                hats[hat] |= 1 << pi;
            }
        }
	
        vector<vector<int>> dp(m + 1, vector<int>(1 << n, -1));
        return f(hats, m, n, 1, 0, dp);
    }

    int f(vector<int>& hats, int m, int n, int i, int s, vector<vector<int>>& dp) {
        if (s == (1 << n) - 1) {
            return 1;
        }
        if (i == m + 1) {
            return 0;
        }
        if (dp[i][s] != -1) {
            return dp[i][s];
        }

        int ans = f(hats, m, n, i + 1, s, dp);
        int cur = hats[i];
        while (cur != 0) {
            int rightOne = cur & -cur;
            if ((s & rightOne) == 0) {
                ans = (ans + f(hats, m, n, i + 1, s | rightOne, dp)) % MOD;
            }
            cur ^= rightOne;
        }

        return dp[i][s] = ans;
    }
};
```
