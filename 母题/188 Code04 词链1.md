#include <bits/stdc++.h>

using namespace std;

// 词链，java版
// 给定n个由小写字母组成的单词，想把所有单词串成一个词链
// 同时要求前一个单词的结尾字符 == 后一个单词的开头字符
// 比如词链，aloha.arachnid.dog.gopher.rat.tiger
// 需要使用每个输入的单词恰好一次，同一单词出现k次就要用k次
// 返回字典序最小的结果，注意 . 的字典序 < 小写字母的字典序
// 如果不存在合法词链，输出***
// 1 <= n <= 1000
// 测试链接 : https://www.luogu.com.cn/problem/P1127
// 提交以下的code，提交时请把类名改成"Main"，可以通过所有测试用例

const int MAXN = 27;
const int MAXM = 1002;
int n = 26;
int m;
string str[MAXM];
int startLetter[MAXM];
int endLetter[MAXM];
int edgeOrder[MAXM];
int head[MAXN];
int nxt[MAXM];
int to[MAXM];
string weight[MAXM];
int cntg;
int cur[MAXN];
int outDeg[MAXN];
int inDeg[MAXN];
string path[MAXM];
int cntp;

void addEdge(int u, int v, const string& w) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    weight[cntg] = w;
    head[u] = cntg;
}

int startNode(const string& word) {
    return word[0] - 'a' + 1;
}

int endNode(const string& word) {
    return word.back() - 'a' + 1;
}

void connectGraph() {
    sort(edgeOrder + 1, edgeOrder + m + 1, [](int i, int j) {
        if (startLetter[i] != startLetter[j]) {
            return startLetter[i] < startLetter[j];
        }
        return str[i] < str[j];
    });
    for (int l = 1, r = 1; l <= m; l = ++r) {
        while (r + 1 <= m && startLetter[edgeOrder[l]] == startLetter[edgeOrder[r + 1]]) {
            r++;
        }
        for (int i = r; i >= l; i--) {
            int id = edgeOrder[i];
            outDeg[startLetter[id]]++;
            inDeg[endLetter[id]]++;
            addEdge(startLetter[id], endLetter[id], str[id]);
        }
    }
    for (int i = 1; i <= n; i++) {
        cur[i] = head[i];
    }
}

int directedStart() {
    int start = -1;
    int finish = -1;
    for (int i = 1; i <= n; i++) {
        int value = outDeg[i] - inDeg[i];
        if (value < -1 || value > 1 || (value == 1 && start != -1) || (value == -1 && finish != -1)) {
            return -1;
        }
        if (value == 1) {
            start = i;
        }
        if (value == -1) {
            finish = i;
        }
    }
    if ((start == -1) ^ (finish == -1)) {
        return -1;
    }
    if (start != -1) {
        return start;
    }
    for (int i = 1; i <= n; i++) {
        if (outDeg[i] > 0) {
            return i;
        }
    }
    return -1;
}

void euler(int u, const string& w) {
    for (int e = cur[u]; e > 0; e = cur[u]) {
        cur[u] = nxt[e];
        euler(to[e], weight[e]);
    }
    path[++cntp] = w;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> m;
    for (int i = 1; i <= m; i++) {
        cin >> str[i];
        startLetter[i] = startNode(str[i]);
        endLetter[i] = endNode(str[i]);
        edgeOrder[i] = i;
    }
    connectGraph();
    int start = directedStart();
    if (start == -1) {
        cout << "***\n";
    } else {
        euler(start, "");
        if (cntp != m + 1) {
            cout << "***\n";
        } else {
            cout << path[cntp - 1];
            for (int i = cntp - 2; i >= 1; i--) {
                cout << "." << path[i];
            }
            cout << '\n';
        }
    }
    return 0;
}
