#include <bits/stdc++.h>
using namespace std;
// 砍树
// 一共有n棵树，每棵树都有两个信息：第一天这棵树的初始重量、这棵树每天的增长重量
// 你每天最多能砍1棵树，砍下这棵树的收益为：这棵树的初始重量 + 这棵树增长到这一天的总增重
// 从第1天开始，你一共有m天可以砍树，返回m天内你获得的最大收益
// 测试链接 : https://pintia.cn/problem-sets/91827364500/exam/problems/type/7?problemSetProblemId=91827367873
const int MAXN = 251; int tree[MAXN][2], dp[MAXN][MAXN], t, n, m;
// 讲解073 - 01背包
int compute() { sort(tree + 1, tree + n + 1, [](const auto& a, const auto& b) { return a[1] < b[1]; }); for (int i = 1; i <= n; i++) for (int j = 1; j <= m; j++) dp[i][j] = max(dp[i - 1][j], dp[i - 1][j - 1] + tree[i][0] + tree[i][1] * (j - 1)); return dp[n][m]; }
int main() { ios::sync_with_stdio(false); cin.tie(nullptr); cin >> t; while (t--) { cin >> n >> m; for (int i = 1; i <= n; i++) cin >> tree[i][0]; for (int i = 1; i <= n; i++) cin >> tree[i][1]; cout << compute() << '\n'; } return 0; }
