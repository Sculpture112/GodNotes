#include <bits/stdc++.h>
using namespace std;
// 使数组相似的最少操作次数
// 测试链接 : https://leetcode.cn/problems/minimum-number-of-operations-to-make-arrays-similar/
class Solution {
public:
    int split(vector<int>& arr) {
        int oddSize = 0;
        for (int i = 0; i < arr.size(); i++) if (arr[i] & 1) swap(arr[i], arr[oddSize++]);
        return oddSize;
    }
    long long makeSimilar(vector<int>& nums, vector<int>& target) {
        int n = nums.size(), oddSize = split(nums);
        split(target);
        sort(nums.begin(), nums.begin() + oddSize);
        sort(nums.begin() + oddSize, nums.end());
        sort(target.begin(), target.begin() + oddSize);
        sort(target.begin() + oddSize, target.end());
        long long ans = 0;
        for (int i = 0; i < n; i++) ans += abs(1LL * nums[i] - target[i]);
        return ans / 4;
    }
};
