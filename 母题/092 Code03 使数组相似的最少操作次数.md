```cpp
#include <bits/stdc++.h>
using namespace std;
// 使数组相似的最少操作次数
// 给你两个正整数数组 nums 和 target ，两个数组长度相等
// 在一次操作中，你可以选择两个 不同 的下标 i 和 j
// 其中 0 <= i, j < nums.length ，并且：
// 令 nums[i] = nums[i] + 2 且
// 令 nums[j] = nums[j] - 2
// 如果两个数组中每个元素出现的频率相等，我们称两个数组是 相似 的
// 请你返回将 nums 变得与 target 相似的最少操作次数
// 测试数据保证nums一定能变得与target相似
// 测试链接 : https://leetcode.cn/problems/minimum-number-of-operations-to-make-arrays-similar/
class Solution{public:long long makeSimilar(vector<int>&nums,vector<int>&target){auto split=[](vector<int>&a){stable_partition(a.begin(),a.end(),[](int x){return x&1;});};split(nums);split(target);int p=count_if(nums.begin(),nums.end(),[](int x){return x&1;});sort(nums.begin(),nums.begin()+p);sort(nums.begin()+p,nums.end());sort(target.begin(),target.begin()+p);sort(target.begin()+p,target.end());long long ans=0;for(int i=0;i<(int)nums.size();i++)ans+=llabs((long long)nums[i]-target[i]);return ans/4;}};
```
