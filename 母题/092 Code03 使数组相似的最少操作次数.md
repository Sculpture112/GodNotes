#include <bits/stdc++.h>
using namespace std;
// 使数组相似的最少操作次数
// 测试链接 : https://leetcode.cn/problems/minimum-number-of-operations-to-make-arrays-similar/
class Solution{public:int split(vector<int>&a){int k=0;for(int i=0;i<a.size();i++)if(a[i]&1)swap(a[i],a[k++]);return k;}long long makeSimilar(vector<int>&a,vector<int>&b){int k=split(a);split(b);sort(a.begin(),a.begin()+k);sort(a.begin()+k,a.end());sort(b.begin(),b.begin()+k);sort(b.begin()+k,b.end());long long ans=0;for(int i=0;i<a.size();i++)ans+=abs(1LL*a[i]-b[i]);return ans/4;}};
