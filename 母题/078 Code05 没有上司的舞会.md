// 没有上司的舞会
// 某大学有n个职员，编号为1...n
// 他们之间有从属关系，也就是说他们的关系就像一棵以校长为根的树
// 父结点就是子结点的直接上司
// 现在有个周年庆宴会，宴会每邀请来一个职员都会增加一定的快乐指数 
// 但是如果某个职员的直接上司来参加舞会了
// 那么这个职员就无论如何也不肯来参加舞会了
// 所以请你编程计算邀请哪些职员可以使快乐指数最大
// 返回最大的快乐指数。
// 测试链接 : https://www.luogu.com.cn/problem/P1352
// 本题和讲解037的题目7类似
// 链式链接 : https://leetcode.cn/problems/house-robber-iii/
// 请同学们务必参考如下代码中关于输入、输出的处理
// 这是输入输出处理效率很高的写法
// 提交以下的code，提交时请把类名改成"Main"，可以直接通过
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 6005;

int n;
int happy[MAXN];
vector<int> tree[MAXN];
bool hasBoss[MAXN];

int yes[MAXN]; // yes[u]：u 来，u 子树最大快乐值
int no[MAXN];  // no[u]：u 不来，u 子树最大快乐值

void dfs(int u) {
    yes[u] = happy[u];
    no[u] = 0;

    for (int v : tree[u]) {
        dfs(v);

        // u 来，直接下属 v 不能来
        yes[u] += no[v];

        // u 不来，v 来不来都行
        no[u] += max(yes[v], no[v]);
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n;

    for (int i = 1; i <= n; i++) {
        cin >> happy[i];
    }

    for (int i = 1; i <= n - 1; i++) {
        int low, high;
        cin >> low >> high;

        // high 是 low 的直接上司
        tree[high].push_back(low);
        hasBoss[low] = true;
    }

    int root = 1;
    for (int i = 1; i <= n; i++) {
        if (!hasBoss[i]) {
            root = i;
            break;
        }
    }

    dfs(root);

    cout << max(yes[root], no[root]) << '\n';

    return 0;
}
