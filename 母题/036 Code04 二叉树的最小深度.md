```cpp
#include <bits/stdc++.h>

using namespace std;

// 求二叉树的最大、最小深度
// 测试链接 : https://leetcode.cn/problems/minimum-depth-of-binary-tree/
class Solution {
public:
    int minDepth(TreeNode* root) {
        if (root == nullptr) {
            // 当前的树是空树
            return 0;
        }
        if (root->left == nullptr && root->right == nullptr) {
            // 当前root是叶节点
            return 1;
        }
        int ldeep = INT_MAX;
        int rdeep = INT_MAX;
        if (root->left != nullptr) ldeep = minDepth(root->left);
        if (root->right != nullptr) rdeep = minDepth(root->right);
        return min(ldeep, rdeep) + 1;
    }
};
```
