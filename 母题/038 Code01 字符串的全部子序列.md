#include <bits/stdc++.h>

using namespace std;

// 字符串的全部子序列
// 子序列本身是可以有重复的，只是这个题目要求去重
// 测试链接 : https://www.nowcoder.com/practice/92e6247998294f2c933906fdedbc6e6a

// s[i...]，之前决定的路径path，set收集结果时去重
void f1(const string& s, int i, string& path, unordered_set<string>& set) {
    if (i == s.size()) {
        set.insert(path);
    } else {
        path += s[i]; // 加到路径中去
        f1(s, i + 1, path, set);
        path.pop_back(); // 从路径中移除
        f1(s, i + 1, path, set);
    }
}

vector<string> generatePermutation1(string str) {
    unordered_set<string> set;
    string path;
    f1(str, 0, path, set);
    return {set.begin(), set.end()};
}

void f2(const string& s, int i, string& path, int size, unordered_set<string>& set) {
    if (i == s.size()) {
        set.insert(path.substr(0, size));
    } else {
        path[size] = s[i];
        f2(s, i + 1, path, size + 1, set);
        f2(s, i + 1, path, size, set);
    }
}

vector<string> generatePermutation2(string str) {
    unordered_set<string> set;
    string path(str.size(), 0);
    f2(str, 0, path, 0, set);
    return {set.begin(), set.end()};
}
