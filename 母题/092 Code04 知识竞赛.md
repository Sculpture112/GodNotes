int main(){ios::sync_with_stdio(false);cin.tie(nullptr);int n;while(cin>>n){vector<array<int,2>>a(n);for(auto&x:a)cin>>x[0]>>x[1];sort(a.begin(),a.end(),[](auto&x,auto&y){return abs(x[0]-x[1])<abs(y[0]-y[1]);});int A=a[0][0],B=a[0][1],ans=0;for(int i=1;i<n;i++){ans=max(ans,a[i][0]<=a[i][1]?A+a[i][0]:B+a[i][1]);A=max(A,a[i][0]);B=max(B,a[i][1]);}cout<<ans/2.0<<'\n';}}
