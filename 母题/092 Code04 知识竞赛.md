#include <bits/stdc++.h>
using namespace std;
// 知识竞赛
// 测试链接 : https://www.nowcoder.com/practice/2a9089ea7e5b474fa8f688eae76bc050
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    while (cin >> n) {
        vector<array<int, 2>> nums(n);
        for (auto& cur : nums) cin >> cur[0] >> cur[1];
        sort(nums.begin(), nums.end(), [](const auto& a, const auto& b) {
            return abs(a[0] - a[1]) < abs(b[0] - b[1]);
        });
        int maxA = nums[0][0];
        int maxB = nums[0][1];
        int ans = 0;
        for (int i = 1; i < n; i++) {
            if (nums[i][0] <= nums[i][1]) ans = max(ans, maxA + nums[i][0]);
            else ans = max(ans, maxB + nums[i][1]);
            maxA = max(maxA, nums[i][0]);
            maxB = max(maxB, nums[i][1]);
        }
        cout << (double)ans / 2 << '\n';
    }
    return 0;
}
