#include <bits/stdc++.h>

using namespace std;

// 不停删除之后剩下的字符串
// 给定一个字符串s1，如果其中含有s2字符串，就删除最左出现的那个
// 删除之后s1剩下的字符重新拼接在一起，再删除最左出现的那个
// 如此周而复始，返回最终剩下的字符串
// 测试链接 : https://www.luogu.com.cn/problem/P4824
// 请同学们务必参考如下代码中关于输入、输出的处理
// 这是输入输出处理效率很高的写法
// 提交以下的code，提交时请把类名改成"Main"，可以直接通过

const int MAXN = 1000001;

char s1[MAXN];
char s2[MAXN];
int nextData[MAXN];
int stack1[MAXN];
int stack2[MAXN];
int stackSize;

void nextArray(int m) {
    nextData[0] = -1;
    nextData[1] = 0;
    int i = 2;
    int cn = 0;
    while (i < m) {
        if (s2[i - 1] == s2[cn]) {
            nextData[i++] = ++cn;
        } else if (cn > 0) {
            cn = nextData[cn];
        } else {
            nextData[i++] = 0;
        }
    }
}

void compute(const string& first, const string& second) {
    copy(first.begin(), first.end(), s1);
    copy(second.begin(), second.end(), s2);
    stackSize = 0;
    int n = first.length();
    int m = second.length();
    int x = 0;
    int y = 0;
    nextArray(m);
    while (x < n) {
        if (s1[x] == s2[y]) {
            stack1[stackSize] = x;
            stack2[stackSize] = y;
            stackSize++;
            x++;
            y++;
        } else if (y == 0) {
            stack1[stackSize] = x;
            stack2[stackSize] = -1;
            stackSize++;
            x++;
        } else {
            y = nextData[y];
        }
        if (y == m) {
            // 相当于栈直接弹出了m条记录
            stackSize -= m;
            y = stackSize > 0 ? stack2[stackSize - 1] + 1 : 0;
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    string first;
    string second;
    cin >> first >> second;
    compute(first, second);
    for (int i = 0; i < stackSize; i++) {
        cout << s1[stack1[i]];
    }
    cout << '\n';
    return 0;
}
