#include <bits/stdc++.h>

using namespace std;

// 验证完全二叉树
// 测试链接 : https://leetcode.cn/problems/check-completeness-of-a-binary-tree/
class Solution {
public:
    // 提交以下的方法
    bool isCompleteTree(TreeNode* h) {
        if (h == nullptr) return true;
        queue<TreeNode*> q;
        q.push(h);
        // 是否遇到过左右两个孩子不双全的节点
        bool leaf = false;
        while (!q.empty()) {
            h = q.front();
            q.pop();
            if ((h->left == nullptr && h->right != nullptr) ||
                (leaf && (h->left != nullptr || h->right != nullptr))) {
                return false;
            }
            if (h->left != nullptr) q.push(h->left);
            if (h->right != nullptr) q.push(h->right);
            if (h->left == nullptr || h->right == nullptr) leaf = true;
        }
        return true;
    }
};
