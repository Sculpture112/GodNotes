# [077 Code06 统计不同回文子序列]

> **原题链接:** (https://leetcode.cn/problems/count-different-palindromic-subsequences/)

**题目简介:** 给定字符串 `s`，统计其中有多少个不同的非空回文子序列，结果对 `1000000007` 取模。

**涉及知识点:** [[区间DP]], [[前后缀]]，[[待补题]],[[已AC]],[[]]

**核心套路:** 设 `dp[i][j]` 为区间 `i...j` 内不同回文子序列的个数。若两端字符不同，做容斥；若两端字符相同，则根据内部相同字符出现次数分三类讨论。

## 破题切入点 (思维闪念)
[难点不在“是不是回文”，而在“怎么去重”。所以先预处理每个位置左侧和右侧最近的相同字符，再在区间 DP 里精细分类。]



```cpp
// 077 Code06 统计不同回文子序列

// 给你一个字符串s，返回s中不同的非空回文子序列个数
// 由于答案可能很大，答案对1000000007取模
// 测试链接 : https://leetcode.cn/problems/count-different-palindromic-subsequences/

#include <iostream>
#include <vector>
#include <string>
#include <algorithm>

using namespace std;

class Solution {
public:
    static const int MOD = 1000000007;

    int countPalindromicSubsequences(string s) {
        int n = s.size();
        if (n == 0) return 0;

        vector<int> left(n), right(n);

        // left[i]：i 左边最近的相同字符位置，没有就是 -1
        vector<int> last(26, -1);
        for (int i = 0; i < n; i++) {
            int c = s[i] - 'a';
            left[i] = last[c];
            last[c] = i;
        }

        // right[i]：i 右边最近的相同字符位置，没有就是 n
        fill(last.begin(), last.end(), n);
        for (int i = n - 1; i >= 0; i--) {
            int c = s[i] - 'a';
            right[i] = last[c];
            last[c] = i;
        }

        vector<vector<long long>> dp(n, vector<long long>(n, 0));

        for (int i = 0; i < n; i++) {
            dp[i][i] = 1;
        }

        for (int i = n - 2; i >= 0; i--) {
            for (int j = i + 1; j < n; j++) {
                if (s[i] != s[j]) {
                    dp[i][j] =
                        dp[i][j - 1]
                        + dp[i + 1][j]
                        - dp[i + 1][j - 1];
                } else {
                    int l = right[i];
                    int r = left[j];

                    if (l > r) {
                        // 中间没有和 s[i] 相同的字符
                        dp[i][j] = dp[i + 1][j - 1] * 2 + 2;
                    } else if (l == r) {
                        // 中间恰好一个和 s[i] 相同的字符
                        dp[i][j] = dp[i + 1][j - 1] * 2 + 1;
                    } else {
                        // 中间至少两个和 s[i] 相同的字符
                        dp[i][j] =
                            dp[i + 1][j - 1] * 2
                            - dp[l + 1][r - 1];
                    }
                }

                dp[i][j] = (dp[i][j] % MOD + MOD) % MOD;
            }
        }

        return dp[0][n - 1];
    }
};
```
