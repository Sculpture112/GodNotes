#include <bits/stdc++.h>

using namespace std;

// 加入差值绝对值直到长度固定
// 给定一个非负数组arr，计算任何两个数差值的绝对值
// 如果arr中没有，都要加入到arr里，但是只加一份
// 然后新的arr继续计算任何两个数差值的绝对值，
// 如果arr中没有，都要加入到arr里，但是只加一份
// 一直到arr大小固定，返回arr最终的长度
// 来自真实大厂笔试，没有在线测试，对数器验证

bool finish(vector<int>& list, unordered_set<int>& set);

// 暴力方法
// 为了验证
int len1(vector<int>& arr) {
    vector<int> list;
    unordered_set<int> set;
    for (int num : arr) {
        list.push_back(num);
        set.insert(num);
    }
    while (!finish(list, set)) {
    }
    return list.size();
}

bool finish(vector<int>& list, unordered_set<int>& set) {
    int len = list.size();
    for (int i = 0; i < len; i++) {
        for (int j = i + 1; j < len; j++) {
            int absValue = abs(list[i] - list[j]);
            if (!set.contains(absValue)) {
                list.push_back(absValue);
                set.insert(absValue);
            }
        }
    }
    return len == list.size();
}

int gcd(int m, int n) {
    return n == 0 ? m : gcd(n, m % n);
}

// 正式方法
// 时间复杂度O(n)
int len2(vector<int>& arr) {
    int maxValue = 0;
    // 找到任意一个非0的值
    int g = 0;
    for (int num : arr) {
        maxValue = max(maxValue, num);
        if (num != 0) {
            g = num;
        }
    }
    if (g == 0) { // 数组中都是0
        return arr.size();
    }
    // 不都是0
    // 0 7次
    // 5 5次
    unordered_map<int, int> cnts;
    for (int num : arr) {
        if (num != 0) {
            g = gcd(g, num);
        }
        cnts[num]++;
    }
    int ans = maxValue / g;
    int maxCnt = 0;
    for (auto& [key, value] : cnts) {
        if (key != 0) {
            ans += value - 1;
        }
        maxCnt = max(maxCnt, value);
    }
    ans += cnts.contains(0) ? cnts[0] : (maxCnt > 1 ? 1 : 0);
    return ans;
}
