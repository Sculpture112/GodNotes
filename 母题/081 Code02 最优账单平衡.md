#include <bits/stdc++.h>

using namespace std;

// 最优账单平衡
// 给你一个表示交易的数组 transactions
// 其中 transactions[i] = [fromi, toi, amounti]
// 表示 ID = fromi 的人给 ID = toi 的人共计 amounti
// 请你计算并返回还清所有债务的最小交易笔数
// 测试链接 : https://leetcode.cn/problems/optimal-account-balancing/
class Code02_OptimalAccountBalancing {
public:
    // 题目说了人员编号的最大范围：0 ~ 12
    int MAXN = 13;

    int minTransfers(vector<vector<int>>& transactions) {
        // 加工出来的debt数组中一定不含有0
        vector<int> debt = debts(transactions);
        int n = debt.size();
        vector<int> dp(1 << n, -1);
        return n - f(debt, (1 << n) - 1, 0, n, dp);
    }

    vector<int> debts(vector<vector<int>>& transactions) {
        vector<int> help(MAXN, 0);
        for (vector<int>& tran : transactions) {
            help[tran[0]] -= tran[2];
            help[tran[1]] += tran[2];
        }
        vector<int> debt;
        for (int num : help) {
            if (num != 0) {
                debt.push_back(num);
            }
        }
        return debt;
    }

    int f(vector<int>& debt, int set, int sum, int n, vector<int>& dp) {
        if (dp[set] != -1) {
            return dp[set];
        }
        int ans = 0;
        if ((set & (set - 1)) != 0) { // 集合中不只一个元素
            if (sum == 0) {
                for (int i = 0; i < n; i++) {
                    if ((set & (1 << i)) != 0) {
                        // 找到任何一个元素，去除这个元素
                        // 剩下的集合进行尝试，返回值 + 1
                        ans = f(debt, set ^ (1 << i), sum - debt[i], n, dp) + 1;
                        // 然后不需要再尝试下一个元素了，因为答案一定是一样的
                        // 所以直接break
                        break;
                    }
                }
            } else {
                for (int i = 0; i < n; i++) {
                    if ((set & (1 << i)) != 0) {
                        ans = max(ans, f(debt, set ^ (1 << i), sum - debt[i], n, dp));
                    }
                }
            }
        }
        dp[set] = ans;
        return ans;
    }
};
