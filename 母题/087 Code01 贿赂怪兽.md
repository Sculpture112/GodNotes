#include <bits/stdc++.h>
using namespace std;
// 贿赂怪兽
// 开始时你的能力是0，你的目标是从0号怪兽开始，通过所有的n只怪兽
// 如果你当前的能力小于i号怪兽的能力，则必须付出b[i]的钱贿赂这个怪兽
// 然后怪兽就会加入你，他的能力a[i]直接累加到你的能力上
// 如果你当前的能力大于等于i号怪兽的能力，你可以选择直接通过，且能力不会下降
// 但你依然可以选择贿赂这个怪兽，然后怪兽的能力直接累加到你的能力上
// 返回通过所有的怪兽，需要花的最小钱数
// 测试链接 : https://www.nowcoder.com/practice/736e12861f9746ab8ae064d4aae2d5a9
int compute(int n, vector<int>& a, vector<int>& b) {
    int m = accumulate(b.begin() + 1, b.end(), 0);
    vector<int> dp(m + 1, INT_MIN);
    dp[0] = 0;
    for (int i = 1; i <= n; i++) for (int j = m; j >= 0; j--) {
        int cur = dp[j] >= a[i] ? dp[j] : INT_MIN;
        if (j >= b[i] && dp[j - b[i]] != INT_MIN) cur = max(cur, dp[j - b[i]] + a[i]);
        dp[j] = cur;
    }
    for (int j = 0; j <= m; j++) if (dp[j] != INT_MIN) return j;
    return -1;
}
int main() { ios::sync_with_stdio(false); cin.tie(nullptr); int n; while (cin >> n) { vector<int>a(n+1),b(n+1); for(int i=1;i<=n;i++) cin>>a[i]>>b[i]; cout<<compute(n,a,b)<<'\n'; } return 0; }
