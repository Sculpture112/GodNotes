#include <bits/stdc++.h>

using namespace std;

// 无向图的欧拉路径，java版
// 图中给定m条无向边，每条边给出两个端点
// 如果存在欧拉路径，输出字典序最小的结果，如果不存在打印No
// 节点编号的范围是[1, 500]，所有点不一定都出现，以给定的边为准
// 1 <= m <= 1024
// 测试链接 : https://www.luogu.com.cn/problem/P2731
// 提交以下的code，提交时请把类名改成"Main"，可以通过所有测试用例

const int MAXN = 501;
const int MAXM = 2001;

struct Edge {
    int u;
    int v;
    int id;
};

int n = 500;
int m;
int k;
Edge edgeArr[MAXM << 1];
int head[MAXN];
int nxt[MAXM << 1];
int to[MAXM << 1];
int edgeId[MAXM << 1];
int cntg;
int cur[MAXN];
int deg[MAXN];
bool vis[MAXM];
int path[MAXM];
int cntp;
int sta[MAXM];
int top;

void addEdge(int u, int v, int id) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    edgeId[cntg] = id;
    head[u] = cntg;
}

void connectGraph() {
    sort(edgeArr + 1, edgeArr + k + 1, [](const Edge& e1, const Edge& e2) {
        if (e1.u != e2.u) {
            return e1.u < e2.u;
        }
        return e1.v < e2.v;
    });
    for (int l = 1, r = 1; l <= k; l = ++r) {
        while (r + 1 <= k && edgeArr[l].u == edgeArr[r + 1].u) {
            r++;
        }
        for (int i = r; i >= l; i--) {
            deg[edgeArr[i].u]++;
            addEdge(edgeArr[i].u, edgeArr[i].v, edgeArr[i].id);
        }
    }
    for (int i = 1; i <= n; i++) {
        cur[i] = head[i];
    }
}

// 无向图中找到一个起点，去生成欧拉回路 或者 欧拉路径
int undirectedStart() {
    int odd = 0;
    for (int i = 1; i <= n; i++) {
        if ((deg[i] & 1) == 1) {
            odd++;
        }
    }
    if (odd != 0 && odd != 2) {
        return -1;
    }
    for (int i = 1; i <= n; i++) {
        if (odd == 0 && deg[i] > 0) {
            return i;
        }
        if (odd == 2 && (deg[i] & 1) == 1) {
            return i;
        }
    }
    return -1;
}

// Hierholzer算法递归版
void euler1(int u) {
    for (int e = cur[u]; e > 0; e = cur[u]) {
        cur[u] = nxt[e];
        if (!vis[edgeId[e]]) {
            vis[edgeId[e]] = true;
            euler1(to[e]);
        }
    }
    path[++cntp] = u;
}

// Hierholzer算法迭代版
void euler2(int node) {
    top = 0;
    sta[++top] = node;
    while (top > 0) {
        int u = sta[top--];
        int e = cur[u];
        if (e != 0) {
            cur[u] = nxt[e];
            if (!vis[edgeId[e]]) {
                vis[edgeId[e]] = true;
                sta[++top] = u;
                sta[++top] = to[e];
            } else {
                sta[++top] = u;
            }
        } else {
            path[++cntp] = u;
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> m;
    k = 0;
    for (int i = 1, u, v; i <= m; i++) {
        cin >> u >> v;
        edgeArr[++k] = {u, v, i};
        edgeArr[++k] = {v, u, i};
    }
    connectGraph();
    int start = undirectedStart();
    if (start == -1) {
        cout << "No\n";
    } else {
        // euler1(start);
        euler2(start);
        if (cntp != m + 1) {
            cout << "No\n";
        } else {
            for (int i = cntp; i >= 1; i--) {
                cout << path[i] << '\n';
            }
        }
    }
    return 0;
}
