#include <bits/stdc++.h>
using namespace std;

// 矩阵乘法
// 矩阵快速幂
// 矩阵快速幂解决斐波那契第n项的问题

vector<vector<int>> multiply(const vector<vector<int>>& a, const vector<vector<int>>& b) {
    int n = a.size();
    int m = b[0].size();
    int k = a[0].size();
    vector<vector<int>> ans(n, vector<int>(m, 0));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            for (int c = 0; c < k; c++) {
                ans[i][j] += a[i][c] * b[c][j];
            }
        }
    }
    return ans;
}

vector<vector<int>> power(vector<vector<int>> m, int p) {
    int n = m.size();
    vector<vector<int>> ans(n, vector<int>(n, 0));
    for (int i = 0; i < n; i++) {
        ans[i][i] = 1;
    }
    while (p != 0) {
        if ((p & 1) != 0) {
            ans = multiply(ans, m);
        }
        m = multiply(m, m);
        p >>= 1;
    }
    return ans;
}

void print(const vector<vector<int>>& m) {
    for (int i = 0; i < (int)m.size(); i++) {
        for (int j = 0; j < (int)m[0].size(); j++) {
            cout << m[i][j] << (j + 1 == (int)m[0].size() ? '\n' : ' ');
        }
    }
}

void f1() {
    vector<vector<int>> a = {{1, 3}, {4, 2}};
    vector<vector<int>> b = {{2, 3}, {3, 2}};
    print(multiply(a, b));
    cout << "======\n";

    vector<vector<int>> c = {{2, 4}, {3, 2}};
    vector<vector<int>> d = {{2, 3, 2}, {3, 2, 3}};
    print(multiply(c, d));
    cout << "======\n";

    vector<vector<int>> e = {{2, 4}, {1, 2}, {3, 1}};
    vector<vector<int>> f = {{2, 3}, {4, 1}};
    print(multiply(e, f));
    cout << "======\n";

    vector<vector<int>> g = {{3, 1, 2}};
    vector<vector<int>> h = {{1, 2, 1}, {3, 2, 1}, {4, 2, -2}};
    print(multiply(g, h));
}

void f2() {
    vector<vector<int>> a = {{1, 2}, {3, 4}};
    vector<vector<int>> b = multiply(a, multiply(a, multiply(a, multiply(a, a))));
    print(b);
    cout << "======\n";
    print(power(a, 5));
}

void f3() {
    vector<vector<int>> start = {{1, 0}};
    vector<vector<int>> m = {{1, 1}, {1, 0}};
    auto a = multiply(start, m);
    print(a);
    cout << "======\n";
    auto b = multiply(a, m);
    print(b);
    cout << "======\n";
    auto c = multiply(b, m);
    print(c);
    cout << "======\n";
    auto d = multiply(c, m);
    print(d);
}

void f4() {
    vector<vector<int>> start = {{1, 0}};
    vector<vector<int>> m = {{1, 1}, {1, 0}};
    print(multiply(start, power(m, 1)));
    cout << "======\n";
    print(multiply(start, power(m, 2)));
    cout << "======\n";
    print(multiply(start, power(m, 3)));
    cout << "======\n";
    print(multiply(start, power(m, 4)));
}

int main() {
    cout << "f1() : \n";
    cout << "矩阵乘法展示开始\n";
    f1();
    cout << "矩阵乘法展示结束\n\n";

    cout << "f2() : \n";
    cout << "矩阵快速幂展示开始\n";
    f2();
    cout << "矩阵快速幂展示结束\n\n";

    cout << "f3() : \n";
    cout << "求斐波那契数列第n项\n";
    cout << "用矩阵乘法解决\n";
    cout << "展示开始\n";
    f3();
    cout << "展示结束\n\n";

    cout << "f4() : \n";
    cout << "求斐波那契数列第n项\n";
    cout << "用矩阵快速幂解决\n";
    cout << "展示开始\n";
    f4();
    cout << "展示结束\n";
    return 0;
}
