struct Game{long long k,b,p=0;long long cost(long long x){return x*max(0LL,k*x+b);}long long earn(){return cost(p+1)-cost(p);}};
int main(){ios::sync_with_stdio(false);cin.tie(nullptr);int n,m;cin>>n>>m;priority_queue<pair<long long,int>>q;vector<Game>g(m);for(int i=0;i<m;i++){cin>>g[i].k>>g[i].b;if(g[i].earn()>0)q.push({g[i].earn(),i});}long long ans=0;for(int i=0;i<n&&!q.empty();i++){auto [v,id]=q.top();q.pop();if(v<=0)break;ans+=v;g[id].p++;if(g[id].earn()>0)q.push({g[id].earn(),id});}cout<<ans<<'\n';}
