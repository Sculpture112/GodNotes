#include <bits/stdc++.h>

using namespace std;

// 用类描述实现前缀树。不推荐！
// 测试链接 : https://leetcode.cn/problems/implement-trie-ii-prefix-tree/

// 路是数组实现的
// 提交时把类名、构造方法改为Trie
class Trie1 {
private:
    struct TrieNode {
        int pass;
        int end;
        TrieNode* nexts[26];

        TrieNode() : pass(0), end(0), nexts{} {
        }
    };

    TrieNode* root;

public:
    Trie1() : root(new TrieNode()) {
    }

    void insert(const string& word) {
        TrieNode* node = root;
        node->pass++;
        for (int i = 0, path; i < word.length(); i++) { // 从左往右遍历字符
            path = word[i] - 'a'; // 由字符，对应成走向哪条路
            if (node->nexts[path] == nullptr) {
                node->nexts[path] = new TrieNode();
            }
            node = node->nexts[path];
            node->pass++;
        }
        node->end++;
    }

    // 如果之前word插入过前缀树，那么此时删掉一次
    // 如果之前word没有插入过前缀树，那么什么也不做
    void erase(const string& word) {
        if (countWordsEqualTo(word) > 0) {
            TrieNode* node = root;
            node->pass--;
            for (int i = 0, path; i < word.length(); i++) {
                path = word[i] - 'a';
                if (--node->nexts[path]->pass == 0) {
                    node->nexts[path] = nullptr;
                    return;
                }
                node = node->nexts[path];
            }
            node->end--;
        }
    }

    // 查询前缀树里，word单词出现了几次
    int countWordsEqualTo(const string& word) {
        TrieNode* node = root;
        for (int i = 0, path; i < word.length(); i++) {
            path = word[i] - 'a';
            if (node->nexts[path] == nullptr) {
                return 0;
            }
            node = node->nexts[path];
        }
        return node->end;
    }

    // 查询前缀树里，有多少单词以pre做前缀
    int countWordsStartingWith(const string& pre) {
        TrieNode* node = root;
        for (int i = 0, path; i < pre.length(); i++) {
            path = pre[i] - 'a';
            if (node->nexts[path] == nullptr) {
                return 0;
            }
            node = node->nexts[path];
        }
        return node->pass;
    }
};

// 路是哈希表实现的
// 提交时把类名、构造方法改为Trie
class Trie2 {
private:
    struct TrieNode {
        int pass;
        int end;
        unordered_map<int, TrieNode*> nexts;

        TrieNode() : pass(0), end(0) {
        }
    };

    TrieNode* root;

public:
    Trie2() : root(new TrieNode()) {
    }

    void insert(const string& word) {
        TrieNode* node = root;
        node->pass++;
        for (int i = 0, path; i < word.length(); i++) { // 从左往右遍历字符
            path = word[i];
            if (!node->nexts.count(path)) {
                node->nexts[path] = new TrieNode();
            }
            node = node->nexts[path];
            node->pass++;
        }
        node->end++;
    }

    void erase(const string& word) {
        if (countWordsEqualTo(word) > 0) {
            TrieNode* node = root;
            TrieNode* nextNode;
            node->pass--;
            for (int i = 0, path; i < word.length(); i++) {
                path = word[i];
                nextNode = node->nexts[path];
                if (--nextNode->pass == 0) {
                    node->nexts.erase(path);
                    return;
                }
                node = nextNode;
            }
            node->end--;
        }
    }

    int countWordsEqualTo(const string& word) {
        TrieNode* node = root;
        for (int i = 0, path; i < word.length(); i++) {
            path = word[i];
            if (!node->nexts.count(path)) {
                return 0;
            }
            node = node->nexts[path];
        }
        return node->end;
    }

    int countWordsStartingWith(const string& pre) {
        TrieNode* node = root;
        for (int i = 0, path; i < pre.length(); i++) {
            path = pre[i];
            if (!node->nexts.count(path)) {
                return 0;
            }
            node = node->nexts[path];
        }
        return node->pass;
    }
};
