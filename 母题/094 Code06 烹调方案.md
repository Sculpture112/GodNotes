```cpp
#include <bits/stdc++.h>
using namespace std;
// 烹调方案
// 课上没有讲这个题，本节课的题目5，砍树问题，现在验证已经很困难了
// 所以找到这个题作为平替，因为贪心的思路是非常类似的
// 一共有n个食材，一共有m秒可以加工食材，你只能串行加工食材
// 每个食材的初始美味度为a[i]，美味衰减速度b[i]，加工花费c[i]秒
// 第i个食材美味度 = a[i] - t * b[i]，t是加工食材的时刻
// 计算m秒内，你能获得的最大美味度总和
// 1 <= n <= 50
// 其他数据 <= 10^5
// 测试链接 : https://www.luogu.com.cn/problem/P1417
struct Food { int a, b, c; };
int n, m; vector<Food> food; vector<long long> dp;
long long compute() { sort(food.begin(), food.end(), [](const Food& x, const Food& y) { return 1LL * x.c * y.b < 1LL * y.c * x.b; }); fill(dp.begin(), dp.end(), -1); dp[0] = 0; for (const Food& f : food) for (int j = m; j >= f.c; j--) if (dp[j - f.c] != -1) dp[j] = max(dp[j], dp[j - f.c] + f.a - 1LL * j * f.b); return *max_element(dp.begin(), dp.end()); }
int main() { ios::sync_with_stdio(false); cin.tie(nullptr); cin >> m >> n; food.resize(n); for (auto& f : food) cin >> f.a; for (auto& f : food) cin >> f.b; for (auto& f : food) cin >> f.c; dp.resize(m + 1); cout << compute() << '\n'; return 0; }
```
