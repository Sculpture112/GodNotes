```cpp
#include <bits/stdc++.h>
using namespace std;
// 最小区间
// 你有k个非递减排列的整数列表
// 找到一个最小区间，使得k个列表中的每个列表至少有一个数包含在其中
// 测试链接 : https://leetcode.cn/problems/smallest-range-covering-elements-from-k-lists/
class Solution{public:vector<int> smallestRange(vector<vector<int>>&nums){using T=tuple<int,int,int>;set<T> s;for(int i=0;i<(int)nums.size();i++)s.insert({nums[i][0],i,0});int bestL=0,bestR=INT_MAX;while(s.size()==nums.size()){auto [mx,mi,mj]=*s.rbegin();auto [mn,ii,jj]=*s.begin();s.erase(s.begin());if(mx-mn<bestR-bestL)bestL=mn,bestR=mx;if(jj+1<(int)nums[ii].size())s.insert({nums[ii][jj+1],ii,jj+1});}return {bestL,bestR};}};
```
