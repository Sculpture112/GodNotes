#include <bits/stdc++.h>
using namespace std;
// 最小区间
// 你有k个非递减排列的整数列表
// 找到一个最小区间，使得k个列表中的每个列表至少有一个数包含在其中
// 测试链接 : https://leetcode.cn/problems/smallest-range-covering-elements-from-k-lists/
class Solution { public: struct Node{int v,i,j;}; struct Cmp{bool operator()(const Node&a,const Node&b)const{return a.v!=b.v?a.v<b.v:a.i<b.i;}}; vector<int> smallestRange(vector<vector<int>>& nums){int k=nums.size(),r=INT_MAX,a=0,b=0;set<Node,Cmp>s;for(int i=0;i<k;i++)s.insert({nums[i][0],i,0});while(s.size()==k){auto mx=*s.rbegin(),mn=*s.begin();s.erase(s.begin());if(mx.v-mn.v<r)r=mx.v-mn.v,a=mn.v,b=mx.v;if(mn.j+1<nums[mn.i].size())s.insert({nums[mn.i][mn.j+1],mn.i,mn.j+1});}return {a,b};} };
