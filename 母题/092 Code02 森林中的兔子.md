```cpp
#include <bits/stdc++.h>
using namespace std;
// 森林中的兔子
// 测试链接 : https://leetcode.cn/problems/rabbits-in-forest/
class Solution{public:int numRabbits(vector<int>&a){sort(a.begin(),a.end());int ans=0;for(int i=0,j=1;i<a.size();j++){int x=a[i];while(j<a.size()&&a[j]==x)j++;ans+=(j-i+x)/(x+1)*(x+1);i=j;}return ans;}};
```
