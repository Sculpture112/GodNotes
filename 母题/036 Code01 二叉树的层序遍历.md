#include <bits/stdc++.h>

using namespace std;

// 二叉树的层序遍历
// 测试链接 : https://leetcode.cn/problems/binary-tree-level-order-traversal/
class Solution {
public:
    // 提交时把方法名改为levelOrder，此方法为普通bfs，此题不推荐
    vector<vector<int>> levelOrder1(TreeNode* root) {
        vector<vector<int>> ans;
        if (root != nullptr) {
            queue<pair<TreeNode*, int>> q;
            q.push({root, 0});
            while (!q.empty()) {
                auto [cur, level] = q.front();
                q.pop();
                if ((int)ans.size() == level) ans.push_back({});
                ans[level].push_back(cur->val);
                if (cur->left != nullptr) q.push({cur->left, level + 1});
                if (cur->right != nullptr) q.push({cur->right, level + 1});
            }
        }
        return ans;
    }

    // 提交时把方法名改为levelOrder，此方法为每次处理一层的优化bfs，此题推荐
    vector<vector<int>> levelOrder(TreeNode* root) {
        vector<vector<int>> ans;
        if (root != nullptr) {
            queue<TreeNode*> q;
            q.push(root);
            while (!q.empty()) {
                int size = q.size();
                vector<int> list;
                for (int i = 0; i < size; i++) {
                    TreeNode* cur = q.front();
                    q.pop();
                    list.push_back(cur->val);
                    if (cur->left != nullptr) q.push(cur->left);
                    if (cur->right != nullptr) q.push(cur->right);
                }
                ans.push_back(list);
            }
        }
        return ans;
    }
};
