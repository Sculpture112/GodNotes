#include <bits/stdc++.h>

using namespace std;

// 加法、减法、乘法的同余原理
// 不包括除法，因为除法必须求逆元，后续课讲述

long long randomNumber() {
    return (1LL * rand() << 31) ^ rand();
}

// 计算 ((a + b) * (c - d) + (a * c - b * d)) % mod 的非负结果
int f1(long long a, long long b, long long c, long long d, int mod) {
    __int128 o1 = a; // a
    __int128 o2 = b; // b
    __int128 o3 = c; // c
    __int128 o4 = d; // d
    __int128 o5 = o1 + o2; // a + b
    __int128 o6 = o3 - o4; // c - d
    __int128 o7 = o1 * o3; // a * c
    __int128 o8 = o2 * o4; // b * d
    __int128 o9 = o5 * o6; // (a + b) * (c - d)
    __int128 o10 = o7 - o8; // (a * c - b * d)
    __int128 o11 = o9 + o10; // ((a + b) * (c - d) + (a * c - b * d))
    // ((a + b) * (c - d) + (a * c - b * d)) % mod
    __int128 o12 = o11 % mod;
    if (o12 < 0) {
        // 如果是负数那么+mod返回
        return (int)(o12 + mod);
    } else {
        // 如果不是负数直接返回
        return (int)o12;
    }
}

// 计算 ((a + b) * (c - d) + (a * c - b * d)) % mod 的非负结果
int f2(long long a, long long b, long long c, long long d, int mod) {
    int o1 = a % mod; // a
    int o2 = b % mod; // b
    int o3 = c % mod; // c
    int o4 = d % mod; // d
    int o5 = (o1 + o2) % mod; // a + b
    int o6 = (o3 - o4 + mod) % mod; // c - d
    int o7 = 1LL * o1 * o3 % mod; // a * c
    int o8 = 1LL * o2 * o4 % mod; // b * d
    int o9 = 1LL * o5 * o6 % mod; // (a + b) * (c - d)
    int o10 = (o7 - o8 + mod) % mod; // (a * c - b * d)
    int ans = (o9 + o10) % mod; // ((a + b) * (c - d) + (a * c - b * d)) % mod
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cout << "测试开始" << '\n';
    int testTime = 100000;
    int mod = 1000000007;
    for (int i = 0; i < testTime; i++) {
        long long a = randomNumber();
        long long b = randomNumber();
        long long c = randomNumber();
        long long d = randomNumber();
        if (f1(a, b, c, d, mod) != f2(a, b, c, d, mod)) cout << "出错了!" << '\n';
    }
    cout << "测试结束" << '\n';

    cout << "===" << '\n';
    long long a = randomNumber();
    long long b = randomNumber();
    long long c = randomNumber();
    long long d = randomNumber();
    cout << "a : " << a << '\n';
    cout << "b : " << b << '\n';
    cout << "c : " << c << '\n';
    cout << "d : " << d << '\n';
    cout << "===" << '\n';
    cout << "f1 : " << f1(a, b, c, d, mod) << '\n';
    cout << "f2 : " << f2(a, b, c, d, mod) << '\n';
    return 0;
}
