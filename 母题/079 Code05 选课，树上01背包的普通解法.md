// 选课，树上01背包的普通解法
// 在大学里每个学生，为了达到一定的学分，必须从很多课程里选择一些课程来学习
// 在课程里有些课程必须在某些课程之前学习，如高等数学总是在其它课程之前学习
// 现在有 N 门功课，每门课有个学分，每门课有一门或没有直接先修课
// 若课程 a 是课程 b 的先修课即只有学完了课程 a，才能学习课程 b
// 一个学生要从这些课程里选择 M 门课程学习
// 问他能获得的最大学分是多少
// 测试链接 : https://www.luogu.com.cn/problem/P2014
// 请同学们务必参考如下代码中关于输入、输出的处理
// 这是输入输出处理效率很高的写法
// 提交以下的code，提交时请把类名改成"Main"，可以直接通过

// 普通解法，邻接表建图 + 相对好懂的动态规划
// 几乎所有题解都是普通解法的思路，只不过优化了常数时间、做了空间压缩
// 但时间复杂度依然是O(n * 每个节点的孩子平均数量 * m的平方)
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 305;

int n, m;
int nums[MAXN];
vector<int> graph[MAXN];

// dp[i][j][k]
// 在 i 节点及其前 j 棵孩子子树里，一共选 k 个节点，且必须选 i，最大收益
vector<vector<int>> dp[MAXN];

void build() {
    for (int i = 0; i <= n; i++) {
        graph[i].clear();
        dp[i].clear();
    }
}

int dfs(int i, int j, int k) {
    if (k == 0) {
        return 0;
    }

    if (j == 0 || k == 1) {
        return nums[i];
    }

    if (dp[i][j][k] != -1) {
        return dp[i][j][k];
    }

    // 第 j 棵孩子子树一个节点都不选
    int ans = dfs(i, j - 1, k);

    // 第 j 棵孩子子树的根
    int v = graph[i][j - 1];

    // 第 j 棵孩子子树选 s 个节点
    for (int s = 1; s < k; s++) {
        ans = max(ans, dfs(i, j - 1, k - s) + dfs(v, (int)graph[v].size(), s));
    }

    dp[i][j][k] = ans;
    return ans;
}

int compute() {
    for (int i = 0; i <= n; i++) {
        int childCnt = graph[i].size();
        dp[i].assign(childCnt + 1, vector<int>(m + 1, -1));
    }

    return dfs(0, (int)graph[0].size(), m);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    while (cin >> n >> m) {
        m++; // 加上虚拟根 0

        build();

        nums[0] = 0;

        for (int i = 1; i <= n; i++) {
            int pre;
            cin >> pre >> nums[i];
            graph[pre].push_back(i);
        }

        cout << compute() << '\n';
    }

    return 0;
}
