#include <bits/stdc++.h>

using namespace std;

// 选课，树上01背包的普通解法
// 在大学里每个学生，为了达到一定的学分，必须从很多课程里选择一些课程来学习
// 在课程里有些课程必须在某些课程之前学习，如高等数学总是在其它课程之前学习
// 现在有 N 门功课，每门课有个学分，每门课有一门或没有直接先修课
// 若课程 a 是课程 b 的先修课即只有学完了课程 a，才能学习课程 b
// 一个学生要从这些课程里选择 M 门课程学习
// 问他能获得的最大学分是多少
// 测试链接 : https://www.luogu.com.cn/problem/P2014
// 请同学们务必参考如下代码中关于输入、输出的处理
// 这是输入输出处理效率很高的写法
// 提交以下的code，提交时请把类名改成"Main"，可以直接通过

// 普通解法，邻接表建图 + 相对好懂的动态规划
// 几乎所有题解都是普通解法的思路，只不过优化了常数时间、做了空间压缩
// 但时间复杂度依然是O(n * 每个节点的孩子平均数量 * m的平方)
class Code05_CourseSelection1 {
public:
    static const int MAXN = 301;

    static int nums[MAXN];
    static vector<int> graph[MAXN];
    static vector<vector<int>> dp[MAXN];
    static int n, m;

    static void build(int n) {
        for (int i = 0; i <= n; i++) {
            graph[i].clear();
            dp[i].clear();
        }
    }

    static int compute() {
        for (int i = 0; i <= n; i++) {
            dp[i].assign((int)graph[i].size() + 1, vector<int>(m + 1, -1));
        }
        return f(0, (int)graph[0].size(), m);
    }

    // 当前来到i号节点为头的子树
    // 只在i号节点、及其i号节点下方的前j棵子树上挑选节点
    // 一共挑选k个节点，并且保证挑选的节点连成一片
    // 返回最大的累加和
    static int f(int i, int j, int k) {
        if (k == 0) {
            return 0;
        }
        if (j == 0 || k == 1) {
            return nums[i];
        }
        if (dp[i][j][k] != -1) {
            return dp[i][j][k];
        }
        int ans = f(i, j - 1, k);
        // 第j棵子树头节点v
        int v = graph[i][j - 1];
        for (int s = 1; s < k; s++) {
            ans = max(ans, f(i, j - 1, k - s) + f(v, (int)graph[v].size(), s));
        }
        dp[i][j][k] = ans;
        return ans;
    }
};

int Code05_CourseSelection1::nums[Code05_CourseSelection1::MAXN];
vector<int> Code05_CourseSelection1::graph[Code05_CourseSelection1::MAXN];
vector<vector<int>> Code05_CourseSelection1::dp[Code05_CourseSelection1::MAXN];
int Code05_CourseSelection1::n = 0;
int Code05_CourseSelection1::m = 0;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    while (cin >> Code05_CourseSelection1::n >> Code05_CourseSelection1::m) {
        // 节点编号从0~n
        Code05_CourseSelection1::m++;
        Code05_CourseSelection1::build(Code05_CourseSelection1::n);
        for (int i = 1, pre; i <= Code05_CourseSelection1::n; i++) {
            cin >> pre >> Code05_CourseSelection1::nums[i];
            Code05_CourseSelection1::graph[pre].push_back(i);
        }
        cout << Code05_CourseSelection1::compute() << '\n';
    }
    return 0;
}
