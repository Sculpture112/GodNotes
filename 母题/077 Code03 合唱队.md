# [077 Code03 合唱队]

> **原题链接:** (https://www.luogu.com.cn/problem/P3205)

**题目简介:** 给定一排同学的身高，每次只能从最左或最右选一个人加入队伍，要求最终出场顺序严格递增，求可行方案数，答案对 `19650827` 取模。

**涉及知识点:** [[区间DP]], [[空间压缩]]，[[待补题]],[[]],[[]]

**核心套路:** 定义区间 `l...r` 已经形成时的方案数，并记录“最后加入的是左端还是右端”。因为每次只能从两端取人，所以状态只依赖更小的相邻区间。

## 破题切入点 (思维闪念)
[每次只从两端取人，就把“还没取的区间”作为状态；再额外记录最后一个入队的人在哪一端，方便判断能否继续严格递增。]



```cpp
// 077 Code03 合唱队

// 每次只能从序列的最左或最右选一个人加入队伍
// 要求最终加入队伍的顺序对应的身高严格递增
// 返回一共有多少种可行方案，答案对19650827取模
// 测试链接 : https://www.luogu.com.cn/problem/P3205

#include <iostream>
#include <algorithm>

using namespace std;

class Code03_HeightAndChoir {
public:
    static const int MAXN = 1001;
    static const int MOD = 19650827;

    static int nums[MAXN];
    static int dp[MAXN][2];
    static int n;

    // 时间复杂度O(n^2)
    // 严格位置依赖的动态规划
    static int compute1() {
        static int fullDp[MAXN][MAXN][2];
        for (int l = 1; l <= n; l++) {
            for (int r = 1; r <= n; r++) {
                fullDp[l][r][0] = 0;
                fullDp[l][r][1] = 0;
            }
        }
        for (int i = 1; i < n; i++) {
            if (nums[i] < nums[i + 1]) {
                fullDp[i][i + 1][0] = 1;
                fullDp[i][i + 1][1] = 1;
            }
        }
        for (int l = n - 2; l >= 1; l--) {
            for (int r = l + 2; r <= n; r++) {
                if (nums[l] < nums[l + 1]) {
                    fullDp[l][r][0] = (fullDp[l][r][0] + fullDp[l + 1][r][0]) % MOD;
                }
                if (nums[l] < nums[r]) {
                    fullDp[l][r][0] = (fullDp[l][r][0] + fullDp[l + 1][r][1]) % MOD;
                }
                if (nums[r] > nums[l]) {
                    fullDp[l][r][1] = (fullDp[l][r][1] + fullDp[l][r - 1][0]) % MOD;
                }
                if (nums[r] > nums[r - 1]) {
                    fullDp[l][r][1] = (fullDp[l][r][1] + fullDp[l][r - 1][1]) % MOD;
                }
            }
        }
        return (fullDp[1][n][0] + fullDp[1][n][1]) % MOD;
    }

    // 时间复杂度O(n^2)
    // 空间压缩
    static int compute2() {
        if (nums[n - 1] < nums[n]) {
            dp[n][0] = 1;
            dp[n][1] = 1;
        } else {
            dp[n][0] = 0;
            dp[n][1] = 0;
        }
        for (int l = n - 2; l >= 1; l--) {
            if (nums[l] < nums[l + 1]) {
                dp[l + 1][0] = 1;
                dp[l + 1][1] = 1;
            } else {
                dp[l + 1][0] = 0;
                dp[l + 1][1] = 0;
            }
            for (int r = l + 2; r <= n; r++) {
                int a = 0;
                int b = 0;
                if (nums[l] < nums[l + 1]) {
                    a = (a + dp[r][0]) % MOD;
                }
                if (nums[l] < nums[r]) {
                    a = (a + dp[r][1]) % MOD;
                }
                if (nums[r] > nums[l]) {
                    b = (b + dp[r - 1][0]) % MOD;
                }
                if (nums[r] > nums[r - 1]) {
                    b = (b + dp[r - 1][1]) % MOD;
                }
                dp[r][0] = a;
                dp[r][1] = b;
            }
        }
        return (dp[n][0] + dp[n][1]) % MOD;
    }
};

int Code03_HeightAndChoir::nums[Code03_HeightAndChoir::MAXN];
int Code03_HeightAndChoir::dp[Code03_HeightAndChoir::MAXN][2];
int Code03_HeightAndChoir::n;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    while (cin >> Code03_HeightAndChoir::n) {
        for (int i = 1; i <= Code03_HeightAndChoir::n; i++) {
            cin >> Code03_HeightAndChoir::nums[i];
        }
        if (Code03_HeightAndChoir::n == 1) {
            cout << 1 << '\n';
        } else {
            cout << Code03_HeightAndChoir::compute2() << '\n';
        }
    }
    return 0;
}
```
