#include <bits/stdc++.h>
using namespace std;
// 雇佣K名工人的最低成本
// 有n名工人，给定两个数组quality和wage
// 其中quality[i]表示第i名工人的工作质量，其最低期望工资为wage[i]
// 给定整数k，返回组成满足上述条件的付费群体所需的最小金额
// 测试链接 : https://leetcode.cn/problems/minimum-cost-to-hire-k-workers/
class Solution { public: struct Employee { double ratio; int quality; }; double mincostToHireWorkers(vector<int>& quality, vector<int>& wage, int k) { vector<Employee> employees; for (int i = 0; i < quality.size(); i++) employees.push_back({1.0 * wage[i] / quality[i], quality[i]}); sort(employees.begin(), employees.end(), [](const auto& a, const auto& b) { return a.ratio < b.ratio; }); priority_queue<int> heap; int qualitySum = 0; double ans = DBL_MAX; for (auto& e : employees) { if (heap.size() < k) { qualitySum += e.quality; heap.push(e.quality); if (heap.size() == k) ans = min(ans, qualitySum * e.ratio); } else if (heap.top() > e.quality) { qualitySum += e.quality - heap.top(); heap.pop(); heap.push(e.quality); ans = min(ans, qualitySum * e.ratio); } } return ans; } };
