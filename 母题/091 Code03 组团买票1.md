#include <bits/stdc++.h>
using namespace std;
// 组团买票
// 景区里一共有m个项目，景区的第i个项目有如下两个参数：game[i] = { Ki, Bi }，Ki、Bi一定是正数
// 单位一共有n个人，每个人最多可以选1个项目来游玩，也可以不选任何项目
// 来自真实大厂笔试，没有在线测试，对数器验证
int f(int i,int n,vector<vector<int>>&g,vector<int>&c){if(i==n){int a=0;for(int j=0;j<g.size();j++)a+=max((g[j][1]-g[j][0]*c[j])*c[j],0);return a;}int a=f(i+1,n,g,c);for(int j=0;j<g.size();j++){c[j]++;a=max(a,f(i+1,n,g,c));c[j]--;}return a;}
int enough1(int n,vector<vector<int>>&g){vector<int>c(g.size());return f(0,n,g,c);}
struct Game{int k,b,p=0;int earn()const{return b-(p+1)*k-p*k;}};struct Cmp{bool operator()(const Game&a,const Game&b)const{return a.earn()<b.earn();}};
int enough2(int n,vector<vector<int>>&g){priority_queue<Game,vector<Game>,Cmp>h;for(auto&x:g)h.push({x[0],x[1]});int a=0;while(n--&&h.top().earn()>0){auto x=h.top();h.pop();a+=x.earn();x.p++;h.push(x);}return a;}
