#include <bits/stdc++.h>
using namespace std;
// 组团买票
// 景区里一共有m个项目，景区的第i个项目有如下两个参数：game[i] = { Ki, Bi }，Ki、Bi一定是正数
// 单位一共有n个人，每个人最多可以选1个项目来游玩，也可以不选任何项目
// 来自真实大厂笔试，没有在线测试，对数器验证
int f(int i, int n, vector<vector<int>>& games, vector<int>& cnts) {
    if (i == n) { int ans = 0; for (int j = 0; j < games.size(); j++) ans += max((games[j][1] - games[j][0] * cnts[j]) * cnts[j], 0); return ans; }
    int ans = f(i + 1, n, games, cnts);
    for (int j = 0; j < games.size(); j++) { cnts[j]++; ans = max(ans, f(i + 1, n, games, cnts)); cnts[j]--; }
    return ans;
}
int enough1(int n, vector<vector<int>>& games) { vector<int> cnts(games.size()); return f(0, n, games, cnts); }
struct Game { int ki, bi, people = 0; int earn() const { return bi - (people + 1) * ki - people * ki; } };
struct Compare { bool operator()(const Game& a, const Game& b) const { return a.earn() < b.earn(); } };
int enough2(int n, vector<vector<int>>& games) { priority_queue<Game, vector<Game>, Compare> heap; for (auto& game : games) heap.push({game[0], game[1]}); int ans = 0; while (n-- && heap.top().earn() > 0) { Game cur = heap.top(); heap.pop(); ans += cur.earn(); cur.people++; heap.push(cur); } return ans; }
