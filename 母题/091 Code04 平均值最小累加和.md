// 平均值最小累加和
// 给定一个数组arr，长度为n，再给定一个数字k，表示一定要将arr划分成k个集合
// 返回每个集合的平均值都累加起来的最小值，平均值向下取整
int f(vector<int>&a,int i,vector<int>&s,vector<int>&c){if(i==a.size()){int ans=0;for(int j=0;j<s.size();j++){if(!c[j])return INT_MAX;ans+=s[j]/c[j];}return ans;}int ans=INT_MAX;for(int j=0;j<s.size();j++){s[j]+=a[i];c[j]++;ans=min(ans,f(a,i+1,s,c));s[j]-=a[i];c[j]--;}return ans;}
int minAverageSum1(vector<int>&a,int k){vector<int>s(k),c(k);return f(a,0,s,c);}int minAverageSum2(vector<int>&a,int k){sort(a.begin(),a.end());int ans=accumulate(a.begin(),a.begin()+k-1,0),sum=accumulate(a.begin()+k-1,a.end(),0);return ans+sum/(a.size()-k+1);}
