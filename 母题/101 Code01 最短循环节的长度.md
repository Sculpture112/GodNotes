#include <bits/stdc++.h>

using namespace std;

// 最短循环节的长度
// 给你一个字符串s，它一定是由某个循环节不断自我连接形成的
// 题目保证至少重复2次，但是最后一个循环节不一定完整
// 现在想知道s的最短循环节是多长
// 测试链接 : https://www.luogu.com.cn/problem/P4391
// 请同学们务必参考如下代码中关于输入、输出的处理
// 这是输入输出处理效率很高的写法
// 提交以下的code，提交时请把类名改成"Main"，可以直接通过

const int MAXN = 1000001;

int nextData[MAXN];
int n;
string s;

void nextArray() {
    nextData[0] = -1;
    nextData[1] = 0;
    int i = 2;
    int cn = 0;
    while (i <= n) {
        if (s[i - 1] == s[cn]) {
            nextData[i++] = ++cn;
        } else if (cn > 0) {
            cn = nextData[cn];
        } else {
            nextData[i++] = 0;
        }
    }
}

int compute() {
    nextArray();
    return n - nextData[n];
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n;
    cin >> s;
    cout << compute() << '\n';
    return 0;
}
