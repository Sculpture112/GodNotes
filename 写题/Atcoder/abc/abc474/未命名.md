#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N;
    cin >> N;

    vector<int64> A(N + 1);
    for (int i = 1; i <= N; ++i) {
        cin >> A[i];
    }

    // C[i] = coef[i] * x + constant[i]
    vector<int64> coef(N + 1);
    vector<int64> constant(N + 1);

    int64 lower = *max_element(A.begin() + 1, A.end());
    int64 upper = 4'000'000'000'000'000'000LL;

    for (int i = N; i >= 1; --i) {
        // C[i] = x - A[i] - C[2i] - C[3i] - ...
        coef[i] = 1;
        constant[i] = -A[i];

        for (int j = i + i; j <= N; j += i) {
            coef[i] -= coef[j];
            constant[i] -= constant[j];
        }

        // coef[i] * x + constant[i] >= 0
        if (coef[i] > 0) {
            if (constant[i] < 0) {
                int64 need =
                    (-constant[i] + coef[i] - 1) / coef[i];
                lower = max(lower, need);
            }
        } else if (coef[i] == 0) {
            if (constant[i] < 0) {
                cout << -1 << '\n';
                return 0;
            }
        } else {
            // coef[i] < 0:
            // x <= constant[i] / (-coef[i])
            if (constant[i] < 0) {
                cout << -1 << '\n';
                return 0;
            }

            upper = min(
                upper,
                constant[i] / (-coef[i])
            );
        }
    }

    if (lower > upper) {
        cout << -1 << '\n';
    } else {
        cout << lower - A[1] << '\n';
    }

    return 0;
}
