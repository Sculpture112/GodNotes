**核心套路:** 
数位和%9 == 0 即可被9 整除

有 n 个位置，位置编号 0..n-1
如果规定 i 和 i+k 必须属于同一类
那么所有位置会被分成 gcd(n, k) 组


## 破题切入点 (思维闪念)
[]



```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int n;
        string s;
        cin >> n >> s;

        int g = gcd(n, 9);
		//先用gcd分组
        vector<vector<int>> cnt(g, vector<int>(9, 0));
        //group组的余数统计
        vector<int> sz(g, 0);
		//group的大小
        for (int i = 0; i < n; i++) {
            int group = i % g;
            int r = (s[i] - '0') % 9;
		
            cnt[group][r]++;
            sz[group]++;
        }

        vector<vector<int>> cost(g, vector<int>(9, 0));
		//gourp组到达余数需要的花费
        for (int group = 0; group < g; group++) {
            for (int r = 0; r < 9; r++) {
                cost[group][r] = sz[group] - cnt[group][r];
                //总的大小减已经是r的数量
            }
        }

        const int INF = 1e9;

        vector<int> dp(9, INF);
        //各组只选一个的余数和为sum的值就是dp的值
        dp[0] = 0;

        for (int group = 0; group < g; group++) {
            vector<int> ndp(9, INF);

            for (int oldSum = 0; oldSum < 9; oldSum++) {
                if (dp[oldSum] == INF) continue;

                for (int r = 0; r < 9; r++) {
                    int newSum = (oldSum + r) % 9;
                    ndp[newSum] = min(
                        ndp[newSum],
                        dp[oldSum] + cost[group][r]
                    );
                }
            }

            dp = ndp;
        }

        int ans = INF;
		//windowsum就是九个字符连起来的余数和
        for (int sum = 0; sum < 9; sum++) {
            int windowSum = (sum * (9 / g)) % 9;

            if (windowSum == 0) {
                ans = min(ans, dp[sum]);
            }
        }

        cout << ans << '\n';
    }

    return 0;
}
```
