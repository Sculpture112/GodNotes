#include <iostream>
#include <set>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int m;
    cin >> m;

    set<long long> wood;

    while (m--) {
        int op;
        long long length;
        cin >> op >> length;

        if (op == 1) {
            // insert 返回值的 second 表示是否成功插入
            if (!wood.insert(length).second) {
                cout << "Already Exist\n";
            }
        } else {
            if (wood.empty()) {
                cout << "Empty\n";
                continue;
            }

            // 第一个 >= length 的木材
            auto right = wood.lower_bound(length);
            auto chosen = right;

            if (right == wood.begin()) {
                // 没有更短的，只能选当前木材
                chosen = right;
            } else if (right == wood.end()) {
                // 没有更长的，只能选最后一根
                chosen = prev(right);
            } else {
                auto left = prev(right);

                // 距离相同时选择较短的，所以使用 <=
                if (length - *left <= *right - length) {
                    chosen = left;
                }
            }

            cout << *chosen << '\n';
            wood.erase(chosen);
        }
    }

    return 0;
}
