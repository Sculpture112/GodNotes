#include <bits/stdc++.h>
using namespace std;

struct Hash {
    size_t operator()(uint64_t x) const {
        x += 0x9e3779b97f4a7c15ULL;
        x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9ULL;
        x = (x ^ (x >> 27)) * 0x94d049bb133111ebULL;
        return x ^ (x >> 31);
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, k, x;
    cin >> n >> k >> x;
    vector<int> a(n);
    for (int &v : a) cin >> v;

    int m = n / 2, r = n - m;
    unordered_map<uint64_t, int, Hash> freq;
    freq.max_load_factor(0.7f);
    freq.reserve(1u << m);

    int leftN = 1 << m;
    vector<int> leftXor(leftN);
    for (int mask = 0; mask < leftN; ++mask) {
        if (mask) {
            int bit = mask & -mask;
            leftXor[mask] = leftXor[mask ^ bit]
                          ^ a[__builtin_ctz((unsigned)bit)];
        }

        int cnt = __builtin_popcount((unsigned)mask);
        if (cnt <= k && k - cnt <= r) {
            uint64_t key = (uint64_t(uint32_t(leftXor[mask])) << 6)
                         | uint64_t(cnt);
            ++freq[key];
        }
    }

    int rightN = 1 << r;
    vector<int> rightXor(rightN);
    long long ans = 0;

    for (int mask = 0; mask < rightN; ++mask) {
        if (mask) {
            int bit = mask & -mask;
            rightXor[mask] = rightXor[mask ^ bit]
                           ^ a[m + __builtin_ctz((unsigned)bit)];
        }

        int cnt = __builtin_popcount((unsigned)mask);
        int needCnt = k - cnt;
        if (needCnt < 0 || needCnt > m) continue;

        uint64_t key = (uint64_t(uint32_t(x ^ rightXor[mask])) << 6)
                     | uint64_t(needCnt);
        auto it = freq.find(key);
        if (it != freq.end()) ans += it->second;
    }

    cout << ans % 1000000007LL << '\n';
    return 0;
}
