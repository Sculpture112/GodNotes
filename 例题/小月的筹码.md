```cpp
#include <bits/stdc++.h>
using namespace std;

const int MOD = 1000000007;

int n, k, x, mid;
int a[40];
unordered_map<int, int> freq[21];
long long ans = 0;

// 枚举左半边，统计每种状态的方案数
void dfsLeft(int pos, int cnt, int val) {
    if (cnt > k) return;

    if (pos == mid) {
        ++freq[cnt][val];
        return;
    }

    dfsLeft(pos + 1, cnt, val);                  // 不选
    dfsLeft(pos + 1, cnt + 1, val ^ a[pos]);     // 选
}

// 枚举右半边，查找能与它匹配的左半边状态
void dfsRight(int pos, int cnt, int val) {
    if (cnt > k) return;

    if (pos == n) {
        int need = k - cnt;
        if (need > mid) return;

        auto it = freq[need].find(x ^ val);
        if (it != freq[need].end()) {
            ans += it->second;
        }
        return;
    }

    dfsRight(pos + 1, cnt, val);                 // 不选
    dfsRight(pos + 1, cnt + 1, val ^ a[pos]);    // 选
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> k >> x;
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }

    mid = n / 2;

    dfsLeft(0, 0, 0);
    dfsRight(mid, 0, 0);

    cout << ans % MOD << '\n';
    return 0;
}
```
