#include <bits/stdc++.h>
using namespace std;

using int64 = long long;
const int64 MOD = 998244353;

struct Block {
    int64 value; // 当前块表示的数字
    int64 place; // 10^(当前块位数)
};

// 把 left 和 right 两个十进制块拼接起来
Block mergeBlock(const Block& left, const Block& right) {
    return {
        (left.value * right.place + right.value) % MOD,
        left.place * right.place % MOD
    };
};

// 计算字符串前 length 位组成的数字及其位数倍数
Block makeBlock(const string& s, int length) {
    Block res{0, 1};

    for (int i = 0; i < length; ++i) {
        res.value = (res.value * 10 + (s[i] - '0')) % MOD;
        res.place = res.place * 10 % MOD;
    }

    return res;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    int64 l, r;
    string x;

    cin >> n >> l >> r;
    cin >> x;

    // 找到从第 l 位开始的旋转周期
    int start = static_cast<int>((l - 1) % n);
    string cycle = x.substr(start) + x.substr(0, start);

    int64 length = r - l + 1;
    int64 fullCount = length / n;
    int remainder = static_cast<int>(length % n);

    // blocks[i] 表示 cycle 重复 2^i 次
    Block blocks[61];
    blocks[0] = makeBlock(cycle, n);

    for (int i = 1; i <= 60; ++i) {
        blocks[i] = mergeBlock(blocks[i - 1], blocks[i - 1]);
    }

    // 按从大到小的方式拼出 fullCount 个完整周期
    Block answer{0, 1};
    int64 remain = fullCount;

    for (int i = 60; i >= 0; --i) {
        int64 count = 1LL << i;

        if (count <= remain) {
            answer = mergeBlock(answer, blocks[i]);
            remain -= count;
        }
    }

    // 接上不足一个周期的部分
    Block tail = makeBlock(cycle, remainder);
    answer = mergeBlock(answer, tail);

    cout << answer.value << '\n';
    return 0;
}
