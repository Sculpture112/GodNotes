#include <bits/stdc++.h>
using namespace std;

using int64 = long long;
const int64 MOD = 998244353;

int64 geometricSum(int64 base, int64 k) {
    int64 resPow = 1; // base^processed
    int64 resSum = 0; // 1 + base + ... + base^(processed-1)

    int64 blockPow = base;
    int64 blockSum = 1;

    while (k > 0) {
        if (k & 1) {
            resSum = (resSum + resPow * blockSum) % MOD;
            resPow = resPow * blockPow % MOD;
        }

        blockSum = (blockSum + blockPow * blockSum) % MOD;
        blockPow = blockPow * blockPow % MOD;
        k >>= 1;
    }

    return resSum;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    int64 l, r;
    string x;

    cin >> n >> l >> r;
    cin >> x;

    vector<int64> pw10(n + 1), prefix(n + 1);
    pw10[0] = 1;

    for (int i = 1; i <= n; ++i) {
        pw10[i] = pw10[i - 1] * 10 % MOD;
    }

    for (int i = 0; i < n; ++i) {
        prefix[i + 1] = (prefix[i] * 10 + (x[i] - '0')) % MOD;
    }

    auto getValue = [&](int L, int R) -> int64 {
        int len = R - L + 1;
        return (prefix[R + 1] -
                prefix[L] * pw10[len] % MOD +
                MOD) % MOD;
    };

    int start = static_cast<int>((l - 1) % n);
    int64 len = r - l + 1;

    int64 full = len / n;
    int64 rem = len % n;

    // 从 start 开始的一个完整周期
    int64 suffix = getValue(start, n - 1);
    int64 front = prefix[start];
    int64 cycle = (suffix * pw10[start] + front) % MOD;

    // 剩余 rem 位
    int64 rest = 0;

    if (rem > 0) {
        if (rem <= n - start) {
            rest = getValue(start, start + static_cast<int>(rem) - 1);
        } else {
            int extra = static_cast<int>(rem - (n - start));
            rest = (suffix * pw10[extra] + prefix[extra]) % MOD;
        }
    }

    int64 base = pw10[n];
    int64 sum = geometricSum(base, full);

    int64 answer = (cycle * sum % MOD * pw10[rem] + rest) % MOD;
    cout << answer << '\n';

    return 0;
}
