#include <bits/stdc++.h>
using namespace std;

using ll = long long;

struct Edge {
    int to;
    ll d, r;
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    vector<vector<Edge>> g(n + 1);
    for (int i = 0; i < m; ++i) {
        int u, v;
        ll d, r;
        cin >> u >> v >> d >> r;
        g[u].push_back({v, d, r});
    }

    const ll INF = 1LL << 62;
    vector<pair<ll, ll>> dist(n + 1, {INF, INF});

    using State = tuple<ll, ll, int>; // 距离、风险、结点
    priority_queue<State, vector<State>, greater<State>> pq;

    dist[1] = {0, 0};
    pq.push({0, 0, 1});

    while (!pq.empty()) {
        auto [d, r, u] = pq.top();
        pq.pop();

        if (make_pair(d, r) != dist[u]) continue;

        for (const auto& e : g[u]) {
            pair<ll, ll> next = {d + e.d, r + e.r};
            if (next < dist[e.to]) {
                dist[e.to] = next;
                pq.push({next.first, next.second, e.to});
            }
        }
    }

    if (dist[n].first == INF)
        cout << "-1 -1\n";
    else
        cout << dist[n].first << ' ' << dist[n].second << '\n';
}
