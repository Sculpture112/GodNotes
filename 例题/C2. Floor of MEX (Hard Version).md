#include <bits/stdc++.h>
using namespace std;

constexpr int MOD = 1'000'000'007;

void solve() {
    int n;
    cin >> n;

    vector<int> a(n + 1);
    for (int k = 1; k <= n; ++k) {
        cin >> a[k];
    }

    vector<pair<int, int>> requiredIntervals;
    vector<pair<int, int>> forbiddenIntervals;

    for (int k = 1; k <= n; ++k) {
        /*
        floor(y / k) 的可能取值为：
        0, 1, ..., floor((n - 1) / k)

        所以 mex 最大只能为 floor((n - 1) / k) + 1。
        题目保证输入合法，因此这个分支实际不会触发。
        */
        if (a[k] > (n - 1) / k + 1) {
            cout << 0 << '\n';
            return;
        }

        // a[k] 不能出现
        int forbiddenLeft = a[k] * k;
        int forbiddenRight = (a[k] + 1) * k - 1;

        forbiddenIntervals.emplace_back(
            forbiddenLeft,
            forbiddenRight
        );

        // 0, 1, ..., a[k]-1 必须出现
        for (int value = 0; value < a[k]; ++value) {
            int left = value * k;
            int right = (value + 1) * k - 1;

            requiredIntervals.emplace_back(left, right);
        }
    }

    /*
    通过差分数组找出所有没有落入禁选区间的数字。
    */
    vector<int> difference(n + 1, 0);

    for (auto [left, right] : forbiddenIntervals) {
        left = min(left, n);
        right = min(right + 1, n);

        ++difference[left];
        --difference[right];
    }

    vector<int> allowedNumbers;
    int cover = 0;

    for (int x = 0; x < n; ++x) {
        cover += difference[x];

        if (cover == 0) {
            allowedNumbers.push_back(x);
        }
    }

    /*
    删除冗余的必须命中区间。

    按长度从小到大处理。如果当前区间包含一个已经保留的
    更小区间，那么命中小区间时一定会命中当前区间。
    */
    sort(
        requiredIntervals.begin(),
        requiredIntervals.end(),
        [](const pair<int, int>& first,
           const pair<int, int>& second) {
            return first.second - first.first
                 < second.second - second.first;
        }
    );

    set<pair<int, int>> importantSet;

    for (auto [left, right] : requiredIntervals) {
        // 寻找左端点 >= left 的第一个已保留区间
        auto it = importantSet.lower_bound({left, 0});
		// [[2026-09-23-08-50-19]]
        // 已经存在一个被当前区间包含的小区间
        if (it != importantSet.end() && it->second <= right) {
            continue;
        }

        importantSet.emplace(left, right);
    }

    /*
    set 按左端点排序。
    删除包含关系后，右端点也会递增。
    */
    vector<pair<int, int>> importantIntervals(
        importantSet.begin(),
        importantSet.end()
    );

    int intervalCount = (int)importantIntervals.size();

    vector<int> dp(intervalCount + 1, 0);
    dp[0] = 1;

    int firstValidState = 0;
    int openedIntervals = 0;
    int possible = 1;

    for (int x : allowedNumbers) {
        /*
        如果第 firstValidState 个区间的右端点已经小于 x，
        那么 dp[firstValidState] 已经不可能命中该区间。
        */
        while (
            firstValidState < intervalCount &&
            importantIntervals[firstValidState].second < x
        ) {
            possible -= dp[firstValidState];

            if (possible < 0) {
                possible += MOD;
            }

            ++firstValidState;
        }

        /*
        找出所有左端点 <= x 的区间。
        x 可以命中这些已经开始的区间。
        */
        while (
            openedIntervals < intervalCount &&
            importantIntervals[openedIntervals].first <= x
        ) {
            ++openedIntervals;
        }

        /*
        选择 x：
        可以把当前所有有效状态转移到 dp[openedIntervals]。
        */
        dp[openedIntervals] += possible;

        if (dp[openedIntervals] >= MOD) {
            dp[openedIntervals] -= MOD;
        }

        /*
        对于每个当前方案：
        1. 不选择 x
        2. 选择 x
        */
        possible = 2LL * possible % MOD;
    }

    cout << dp[intervalCount] << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int testCases;
    cin >> testCases;

    while (testCases--) {
        solve();
    }

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

const int MOD = 1e9 + 7;

void tc() {
    int n; cin >> n;
    vector<int> b(n + 1);
    for(int i = 1; i <= n; i++) cin >> b[i];

    vector<pair<int, int>> goodivs;
    vector<pair<int, int>> badivs;

    for(int i = 1; i <= n; i++) {
        if(b[i] > (n - 1) / i + 1) {
            cout << 0 << endl;
            return;
        }

        badivs.push_back({b[i] * i, (b[i] + 1) * i - 1});

        for(int j = 0; j < b[i]; j++) {
            goodivs.push_back({j * i, (j + 1) * i - 1});
        }
    }

    vector<int> d(n + 1);

    for(auto [l, r] : badivs) {
        d[min(n, l)]++;
        d[min(n, r + 1)]--;
    }

    int csm = 0;
    vector<int> gnums;

    for(int i = 0; i < n; i++) {
        csm += d[i];

        if(csm == 0) {
            gnums.push_back(i);
        }
    }

    set<pair<int, int>> st;

    sort(
        goodivs.begin(),
        goodivs.end(),
        [&](pair<int, int> iv, pair<int, int> iv2) {
            return iv.second - iv.first
                 < iv2.second - iv2.first;
        }
    );

    for(auto [l, r] : goodivs) {
        auto it = st.lower_bound({l, 0});

        if(it != st.end() && it->second <= r) {
            continue;
        }

        st.insert({l, r});
    }

    vector<pair<int, int>> impivs(st.begin(), st.end());

    int l = 0, r = 0;
    vector<int> dp(impivs.size() + 1);

    dp[0] = 1;
    int poss = 1;

    for(int i : gnums) {
        while(l < impivs.size() && impivs[l].second < i) {
            poss += MOD - dp[l];
            poss %= MOD;
            l++;
        }

        while(r < impivs.size() && impivs[r].first <= i) {
            r++;
        }

        dp[r] += poss;
        dp[r] %= MOD;

        poss += poss;
        poss %= MOD;
    }

    cout << dp[impivs.size()] << endl;
}

int main() {
    ios::sync_with_stdio(false), cin.tie(nullptr);

    int t;
    cin >> t;

    while(t--) {
        tc();
    }
}
