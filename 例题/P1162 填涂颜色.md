#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<vector<int>> a(n, vector<int>(n));
    vector<vector<bool>> vis(n, vector<bool>(n, false));
    queue<pair<int, int>> q;

    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            cin >> a[i][j];
        }
    }

    auto add = [&](int x, int y) {
        if (x >= 0 && x < n && y >= 0 && y < n &&
            a[x][y] == 0 && !vis[x][y]) {
            vis[x][y] = true;
            q.push({x, y});
        }
    };

    // 从边界上的 0 开始搜索
    for (int i = 0; i < n; ++i) {
        add(i, 0);
        add(i, n - 1);
    }
    for (int j = 0; j < n; ++j) {
        add(0, j);
        add(n - 1, j);
    }

    int dx[4] = {-1, 1, 0, 0};
    int dy[4] = {0, 0, -1, 1};

    while (!q.empty()) {
        auto [x, y] = q.front();
        q.pop();

        for (int k = 0; k < 4; ++k) {
            add(x + dx[k], y + dy[k]);
        }
    }

    // 未被边界连通的 0 即为圈内区域
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            if (a[i][j] == 0 && !vis[i][j]) {
                a[i][j] = 2;
            }
        }
    }

    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            if (j) cout << ' ';
            cout << a[i][j];
        }
        cout << '\n';
    }

    return 0;
}
