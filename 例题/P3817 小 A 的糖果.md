```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    long long x;
    cin >> n >> x;

    long long ans = 0;
    long long pre = 0;

    for (int i = 0; i < n; ++i) {
        long long cur;
        cin >> cur;

        if (pre + cur > x) {
            long long eat = pre + cur - x;
            ans += eat;
            cur -= eat;
        }

        pre = cur;
    }

    cout << ans << '\n';
    return 0;
}
```
