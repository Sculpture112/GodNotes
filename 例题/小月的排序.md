using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, k;
    cin >> n >> k;

    vector<tuple<int, int, int>> a;
    a.reserve(n);

    for (int i = 0; i < n; ++i) {
        int x;
        cin >> x;
        int ones = __builtin_popcount((unsigned)x);
        int low = (x == 0 ? 31 : __builtin_ctz((unsigned)x));
        a.emplace_back(ones, low, x);
    }

    sort(a.begin(), a.end());
    cout << get<2>(a[k - 1]) << '\n';
    return 0;
}
