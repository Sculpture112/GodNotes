```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, M;
    cin >> N >> M;

    vector<vector<int>> reverseGraph(N + 1);

    for (int i = 0; i < M; ++i) {
        int u, v;
        cin >> u >> v;

        // 原图 u -> v，反图 v -> u
        reverseGraph[v].push_back(u);
    }

    vector<int> answer(N + 1, 0);
    vector<int> st;
    st.reserve(N);

    for (int i = N; i >= 1; --i) {
        if (answer[i] != 0) {
            continue;
        }

        answer[i] = i;
        st.push_back(i);

        while (!st.empty()) {
            int u = st.back();
            st.pop_back();

            for (int v : reverseGraph[u]) {
                if (answer[v] == 0) {
                    answer[v] = i;
                    st.push_back(v);
                }
            }
        }
    }

    for (int i = 1; i <= N; ++i) {
        if (i > 1) cout << ' ';
        cout << answer[i];
    }
    cout << '\n';

    return 0;
}
```
