```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<long long> sour(n), bitter(n);
    for (int i = 0; i < n; ++i) {
        cin >> sour[i] >> bitter[i];
    }

    long long answer = LLONG_MAX;

    // 枚举所有非空食材子集
    for (int mask = 1; mask < (1 << n); ++mask) {
        long long totalSour = 1;
        long long totalBitter = 0;

        for (int i = 0; i < n; ++i) {
            if (mask & (1 << i)) {
                totalSour *= sour[i];
                totalBitter += bitter[i];
            }
        }

        answer = min(answer, llabs(totalSour - totalBitter));
    }

    cout << answer << '\n';
    return 0;
}
```
