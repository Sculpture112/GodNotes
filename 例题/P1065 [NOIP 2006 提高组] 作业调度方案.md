#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int m, n;
    cin >> m >> n;

    int total = m * n;

    // 给定的操作安排顺序
    vector<int> order(total);
    for (int i = 0; i < total; ++i) {
        cin >> order[i];
    }

    // machineOf[j][k]：工件 j 的第 k 道工序使用的机器
    vector<vector<int>> machineOf(n + 1, vector<int>(m));

    // processTime[j][k]：工件 j 的第 k 道工序加工时间
    vector<vector<int>> processTime(n + 1, vector<int>(m));

    for (int j = 1; j <= n; ++j) {
        for (int k = 0; k < m; ++k) {
            cin >> machineOf[j][k];
        }
    }

    for (int j = 1; j <= n; ++j) {
        for (int k = 0; k < m; ++k) {
            cin >> processTime[j][k];
        }
    }

    // schedule[x]：机器 x 上已经安排的区间，按开始时间升序排列
    // 区间形式为 [start, end)
    vector<vector<pair<int, int>>> schedule(m + 1);

    // 每个工件已经安排了多少道工序
    vector<int> cnt(n + 1, 0);

    // 每个工件上一道工序完成的时间
    vector<int> jobEnd(n + 1, 0);

    int answer = 0;

    for (int job : order) {
        int k = cnt[job]++;
        int machine = machineOf[job][k];
        int duration = processTime[job][k];

        auto &intervals = schedule[machine];

        int ready = jobEnd[job];
        int current = ready;
        int start = ready;
        int insertPos = static_cast<int>(intervals.size());

        // 寻找最早可插入的空档
        for (int i = 0; i < static_cast<int>(intervals.size()); ++i) {
            int left = intervals[i].first;
            int right = intervals[i].second;

            // [current, current + duration) 可以放入当前空档
            if (current + duration <= left) {
                start = current;
                insertPos = i;
                break;
            }

            // 当前区间与操作冲突，跳到该区间结束
            current = max(current, right);
        }

        // 后面没有合适的区间，只能接在机器最后
        if (insertPos == static_cast<int>(intervals.size())) {
            start = current;
        }

        int finish = start + duration;

        // 插入机器时间表
        intervals.insert(
            intervals.begin() + insertPos,
            {start, finish}
        );

        jobEnd[job] = finish;
        answer = max(answer, finish);
    }

    cout << answer << '\n';
    return 0;
}
