```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int N, A, B;
    cin >> N >> A >> B;

    vector<int> K(N + 1);
    for (int i = 1; i <= N; ++i) {
        cin >> K[i];
    }

    vector<int> dist(N + 1, -1);
    queue<int> q;

    dist[A] = 0;
    q.push(A);

    while (!q.empty()) {
        int floor = q.front();
        q.pop();

        int up = floor + K[floor];
        int down = floor - K[floor];

        if (up <= N && dist[up] == -1) {
            dist[up] = dist[floor] + 1;
            q.push(up);
        }

        if (down >= 1 && dist[down] == -1) {
            dist[down] = dist[floor] + 1;
            q.push(down);
        }
    }

    cout << dist[B] << '\n';
    return 0;
}
```
