#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int n;
        cin >> n;

        long long mx = 0;
        long long cnt = 0;

        for (int i = 0; i < n; ++i) {
            long long x;
            cin >> x;

            if (x > mx) {
                mx = x;
                cnt = 1;
            } else if (x == mx) {
                ++cnt;
            }
        }

        cout << (cnt <= mx ? "red" : "purple") << '\n';
    }

    return 0;
}
