#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    vector<int> mountain(n);
    for (int i = 0; i < n; ++i) {
        string s;
        cin >> s;
        for (int j = 0; j < m; ++j) {
            if (s[j] == 'H') mountain[i] |= 1 << j;
        }
    }

    vector<int> state, cnt;
    for (int mask = 0; mask < (1 << m); ++mask) {
        if ((mask & (mask << 1)) || (mask & (mask << 2))) continue;
        state.push_back(mask);
        cnt.push_back(__builtin_popcount((unsigned)mask));
    }

    int k = (int)state.size();
    vector<vector<int>> dp(k, vector<int>(k, -1));
    dp[0][0] = 0;

    // dp[a][b]：上一行状态为 a，上上行状态为 b 时的最大数量。
    for (int i = 0; i < n; ++i) {
        vector<vector<int>> ndp(k, vector<int>(k, -1));

        for (int c = 0; c < k; ++c) {
            if (state[c] & mountain[i]) continue;

            for (int a = 0; a < k; ++a) {
                if (state[c] & state[a]) continue;

                for (int b = 0; b < k; ++b) {
                    if (dp[a][b] == -1) continue;
                    if (state[c] & state[b]) continue;

                    ndp[c][a] = max(ndp[c][a], dp[a][b] + cnt[c]);
                }
            }
        }

        dp.swap(ndp);
    }

    int ans = 0;
    for (const auto &row : dp) {
        for (int value : row) ans = max(ans, value);
    }

    cout << ans << '\n';
    return 0;
}
