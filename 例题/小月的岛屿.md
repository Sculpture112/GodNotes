#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    vector<string> grid(n);
    for (auto &row : grid) cin >> row;

    const int dx[8] = {-1, -1, -1, 0, 0, 1, 1, 1};
    const int dy[8] = {-1,  0,  1,-1, 1,-1, 0, 1};

    auto countComponents = [&](bool eightConnected) {
        vector<char> visited(n * m, 0);
        vector<int> q;
        q.reserve(n * m);

        int components = 0, largest = 0;

        for (int start = 0; start < n * m; ++start) {
            int sx = start / m, sy = start % m;
            if (grid[sx][sy] != '1' || visited[start]) continue;

            ++components;
            q.clear();
            q.push_back(start);
            visited[start] = 1;

            for (size_t head = 0; head < q.size(); ++head) {
                int x = q[head] / m;
                int y = q[head] % m;

                for (int d = 0; d < 8; ++d) {
                    if (!eightConnected && abs(dx[d]) + abs(dy[d]) != 1)
                        continue;

                    int nx = x + dx[d], ny = y + dy[d];
                    if (nx < 0 || nx >= n || ny < 0 || ny >= m)
                        continue;

                    int id = nx * m + ny;
                    if (grid[nx][ny] == '1' && !visited[id]) {
                        visited[id] = 1;
                        q.push_back(id);
                    }
                }
            }

            largest = max(largest, (int)q.size());
        }

        return pair<int, int>{components, largest};
    };

    auto [c4, s4] = countComponents(false);
    auto [c8, s8] = countComponents(true);

    cout << c4 - c8 << ' ' << s4 << ' ' << s8 << '\n';
    return 0;
}
