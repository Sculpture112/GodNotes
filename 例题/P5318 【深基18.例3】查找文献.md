```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    vector<vector<int>> graph(n + 1);
    for (int i = 0; i < m; ++i) {
        int x, y;
        cin >> x >> y;
        graph[x].push_back(y);
    }

    for (int i = 1; i <= n; ++i) {
        sort(graph[i].begin(), graph[i].end());
    }

    // DFS
    vector<char> visited(n + 1, false);
    vector<int> dfsOrder;
    vector<pair<int, int>> st;

    visited[1] = true;
    dfsOrder.push_back(1);
    st.push_back({1, 0});

    while (!st.empty()) {
        int u = st.back().first;
        int &idx = st.back().second;

        while (idx < (int)graph[u].size() && visited[graph[u][idx]]) {
            ++idx;
        }

        if (idx == (int)graph[u].size()) {
            st.pop_back();
        } else {
            int v = graph[u][idx++];
            visited[v] = true;
            dfsOrder.push_back(v);
            st.push_back({v, 0});
        }
    }

    for (int x : dfsOrder) cout << x << ' ';
    cout << '\n';

    // BFS
    fill(visited.begin(), visited.end(), false);
    vector<int> bfsOrder;
    queue<int> q;

    visited[1] = true;
    q.push(1);

    while (!q.empty()) {
        int u = q.front();
        q.pop();
        bfsOrder.push_back(u);

        for (int v : graph[u]) {
            if (!visited[v]) {
                visited[v] = true;
                q.push(v);
            }
        }
    }

    for (int x : bfsOrder) cout << x << ' ';
    cout << '\n';

    return 0;
}
```
