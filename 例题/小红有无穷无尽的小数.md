#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    string a, b;
    cin >> a >> b;

    // 兼容可能省略的前导 0
    if ((int)a.size() < n) {
        a = string(n - (int)a.size(), '0') + a;
    }
    if ((int)b.size() < m) {
        b = string(m - (int)b.size(), '0') + b;
    }

    int L = n / gcd(n, m) * m;

    string A(L, '0'), B(L, '0');
    for (int i = 0; i < L; ++i) {
        A[i] = a[i % n];
        B[i] = b[i % m];
    }

    auto simulate = [&](int firstBorrow, string& result) {
        int borrow = firstBorrow;
        result.assign(L, '0');

        for (int i = L - 1; i >= 0; --i) {
            int x = (A[i] - '0') - (B[i] - '0') - borrow;

            if (x < 0) {
                x += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }

            result[i] = char('0' + x);
        }

        // 无限循环要求首尾借位一致
        return borrow == firstBorrow;
    };

    string answer;

    // 优先尝试普通减法；样例中的反向情况会由第二种状态处理
    if (!simulate(0, answer)) {
        simulate(1, answer);
    }

    // 输出整个公共周期，它本身就是合法循环节
    cout << L << '\n';
    cout << answer << '\n';

    return 0;
}
