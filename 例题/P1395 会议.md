#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<vector<int>> graph(n + 1);

    for (int i = 1; i < n; ++i) {
        int a, b;
        cin >> a >> b;
        graph[a].push_back(b);
        graph[b].push_back(a);
    }

    vector<int> parent(n + 1, 0);
    vector<int> depth(n + 1, 0);
    vector<int> size(n + 1, 1);
    vector<int> order;
    order.reserve(n);

    // 迭代 DFS，避免链状树递归过深
    vector<int> stack;
    stack.push_back(1);
    parent[1] = -1;

    long long rootSum = 0;

    while (!stack.empty()) {
        int u = stack.back();
        stack.pop_back();

        order.push_back(u);
        rootSum += depth[u];

        for (int v : graph[u]) {
            if (v == parent[u]) {
                continue;
            }

            parent[v] = u;
            depth[v] = depth[u] + 1;
            stack.push_back(v);
        }
    }

    // 逆序计算子树大小
    for (int i = n - 1; i > 0; --i) {
        int u = order[i];
        size[parent[u]] += size[u];
    }

    // 换根 DP
    vector<long long> distanceSum(n + 1, 0);
    distanceSum[1] = rootSum;

    for (int i = 1; i < n; ++i) {
        int u = order[i];
        int p = parent[u];

        distanceSum[u] =
            distanceSum[p] + n - 2LL * size[u];
    }

    int answerNode = 1;
    long long minDistance = distanceSum[1];

    // 按编号从小到大扫描，严格更小时才更新
    for (int i = 2; i <= n; ++i) {
        if (distanceSum[i] < minDistance) {
            minDistance = distanceSum[i];
            answerNode = i;
        }
    }

    cout << answerNode << ' ' << minDistance << '\n';

    return 0;
}
