#include <bits/stdc++.h>
using namespace std;

using ll = long long;

const int INF = 0x3f3f3f3f;
const ll LINF = 4e18;

#define all(x) (x).begin(), (x).end()

void solve() {
    int n;
    cin >> n;
    vector<int> a(n);
    for(int &x:a){
        cin >> x;
    }
    int ans = a[0];
    for (int i = 1; i < n; i++) {
        if(a[i-1]>=a[i]){
            continue;
        }
        else{
            ans += a[i] - a[i - 1];
        }
    }
    cout << ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) solve();

    return 0;
}
