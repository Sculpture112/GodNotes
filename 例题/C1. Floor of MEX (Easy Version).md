#include <bits/stdc++.h>
using namespace std;

void solve() {
    int n;
    cin >> n;

    vector<int> a(n + 1);
    for (int k = 1; k <= n; ++k) {
        cin >> a[k];
    }

    // diff[x]：位置 x 被多少个禁选区间覆盖
    vector<int> diff(n + 1, 0);

    for (int k = 1; k <= n; ++k) {
        /*
        为了使 mex 等于 a[k]，必须保证不存在 y 满足：
            floor(y / k) == a[k]

        对应的禁选区间为：
            [a[k] * k, (a[k] + 1) * k - 1]
        */
        long long l = 1LL * a[k] * k;
        long long r = 1LL * (a[k] + 1) * k - 1;

        int L = (int)min<long long>(n, l);
        int R = (int)min<long long>(n, r + 1);

        ++diff[L];
        --diff[R];
    }

    vector<int> answer;
    int cover = 0;

    for (int x = 0; x < n; ++x) {
        cover += diff[x];

        // 没有被任何禁选区间覆盖，就把 x 加入 B
        if (cover == 0) {
            answer.push_back(x);
        }
    }

    cout << answer.size() << '\n';
    for (int x : answer) {
        cout << x << ' ';
    }
    cout << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        solve();
    }

    return 0;
}
