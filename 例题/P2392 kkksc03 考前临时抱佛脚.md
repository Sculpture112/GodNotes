```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int s[4];
    for (int i = 0; i < 4; ++i) {
        cin >> s[i];
    }

    int answer = 0;

    for (int subject = 0; subject < 4; ++subject) {
        vector<int> time(s[subject]);
        int sum = 0;

        for (int &x : time) {
            cin >> x;
            sum += x;
        }

        int capacity = sum / 2;
        vector<int> dp(capacity + 1, 0);

        for (int x : time) {
            // 倒序枚举，保证每道题只能分配一次
            for (int j = capacity; j >= x; --j) {
                dp[j] = max(dp[j], dp[j - x] + x);
            }
        }

        answer += sum - dp[capacity];
    }

    cout << answer << '\n';
    return 0;
}
```
