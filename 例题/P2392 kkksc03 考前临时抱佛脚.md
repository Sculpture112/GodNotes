```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int s[4];
    for (int i = 0; i < 4; ++i) {
        cin >> s[i];
    }

    int answer = 0;

    for (int subject = 0; subject < 4; ++subject) {
        vector<int> time(s[subject]);
        int sum = 0;

        for (int &x : time) {
            cin >> x;
            sum += x;
        }

        int capacity = sum / 2;
        vector<int> dp(capacity + 1, 0);

        for (int x : time) {
            // 倒序枚举，保证每道题只能分配一次
            for (int j = capacity; j >= x; --j) {
                dp[j] = max(dp[j], dp[j - x] + x);
            }
        }

        answer += sum - dp[capacity];
    }

    cout << answer << '\n';
    return 0;
}
```

搜索的解如下

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

vector<int> questions;
int target;
int best;

// 枚举当前题是否交给其中一个大脑
void dfs(int index, int currentSum) {
    // 超过总时间的一半，不再继续搜索
    if (currentSum > target) {
        return;
    }

    // 已经凑到一半，这是最优情况
    if (best == target) {
        return;
    }

    if (index == questions.size()) {
        best = max(best, currentSum);
        return;
    }

    // 选择当前题
    dfs(index + 1, currentSum + questions[index]);

    // 不选择当前题，即交给另一个大脑
    dfs(index + 1, currentSum);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int s[4];
    for (int i = 0; i < 4; ++i) {
        cin >> s[i];
    }

    int answer = 0;

    for (int subject = 0; subject < 4; ++subject) {
        questions.resize(s[subject]);

        int total = 0;
        for (int &time : questions) {
            cin >> time;
            total += time;
        }

        target = total / 2;
        best = 0;

        dfs(0, 0);

        // 一边是 best，另一边是 total - best
        // 后者不小于前者，因此它就是本科完成时间
        answer += total - best;
    }

    cout << answer << '\n';
    return 0;
}
```
