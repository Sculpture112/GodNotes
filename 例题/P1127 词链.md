#include <bits/stdc++.h>
using namespace std;

struct Edge {
    string word;
    int to;
};

vector<Edge> graph[26];
int usedIndex[26];
vector<string> path;

void dfs(int u) {
    while (usedIndex[u] < (int)graph[u].size()) {
        Edge e = graph[u][usedIndex[u]++];
        dfs(e.to);
        path.push_back(e.word);
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<int> in(26, 0), out(26, 0);

    for (int i = 0; i < n; i++) {
        string s;
        cin >> s;

        int u = s.front() - 'a';
        int v = s.back() - 'a';

        graph[u].push_back({s, v});
        out[u]++;
        in[v]++;
    }

    int start = -1;
    int plus = 0, minus = 0;

    for (int i = 0; i < 26; i++) {
        int d = out[i] - in[i];

        if (d == 1) {
            plus++;
            start = i;
        } else if (d == -1) {
            minus++;
        } else if (d != 0) {
            cout << "***\n";
            return 0;
        }
    }

    if (!((plus == 1 && minus == 1) || (plus == 0 && minus == 0))) {
        cout << "***\n";
        return 0;
    }

    // 欧拉回路时，选择字母序最小的起点
    if (start == -1) {
        for (int i = 0; i < 26; i++) {
            if (!graph[i].empty()) {
                start = i;
                break;
            }
        }
    }

    // 逆序排序，配合 Hierholzer 算法得到字典序最小结果
    for (int i = 0; i < 26; i++) {
        sort(graph[i].begin(), graph[i].end(),
             [](const Edge& a, const Edge& b) {
                 if (a.word != b.word) return a.word > b.word;
                 return a.to > b.to;
             });
    }

    dfs(start);

    if ((int)path.size() != n) {
        cout << "***\n";
        return 0;
    }

    reverse(path.begin(), path.end());

    for (int i = 0; i < n; i++) {
        if (i) cout << '.';
        cout << path[i];
    }
    cout << '\n';

    return 0;
}
