```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 27;
const int MAXM = 1005;

int n = 26, m;

string str[MAXM];
int a[MAXM], b[MAXM];
int eidArr[MAXM];

int head[MAXN];
int nxt[MAXM];
int to[MAXM];
string weight[MAXM];
int cntg;

int cur[MAXN];
int inDeg[MAXN], outDeg[MAXN];

string path[MAXM];
int cntp;

bool edgeCmp(int i, int j) {
    if (a[i] != a[j]) {
        return a[i] < a[j];
    }
    return str[i] < str[j];
}

void addEdge(int u, int v, const string& w) {
    nxt[++cntg] = head[u];
    to[cntg] = v;
    weight[cntg] = w;
    head[u] = cntg;
}

int startNode(const string& s) {
    return s.front() - 'a' + 1;
}

int endNode(const string& s) {
    return s.back() - 'a' + 1;
}

void connect() { // 数据离散化
    sort(eidArr + 1, eidArr + m + 1, edgeCmp);

    for (int l = 1, r = 1; l <= m; l = ++r) {// 分组头插
        while (r + 1 <= m && a[eidArr[r + 1]] == a[eidArr[l]]) {
            r++;
        }

        // 倒着插入，保证邻接边按字典序从小到大访问
        for (int i = r; i >= l; i--) {
            int id = eidArr[i]; // 现在的第i名,对应原数组的第eidarr[i]名
            // id : 原始下标
            int u = a[id];
            int v = b[id];

            outDeg[u]++;
            inDeg[v]++;
            addEdge(u, v, str[id]);
        }
    }

    for (int i = 1; i <= n; i++) {
        cur[i] = head[i];
    }
}

int directedStart() {
    int start = -1;
    int finish = -1;

    for (int i = 1; i <= n; i++) {
        int diff = outDeg[i] - inDeg[i];

        if (diff < -1 || diff > 1) {
            return -1;
        }

        if (diff == 1) {
            if (start != -1) return -1;
            start = i;
        }

        if (diff == -1) {
            if (finish != -1) return -1;
            finish = i;
        }
    }

    // 只能是欧拉路径或欧拉回路
    if ((start == -1) != (finish == -1)) {
        return -1;
    }

    // 欧拉路径
    if (start != -1) {
        return start;
    }

    // 欧拉回路，选择字典序最小的起点
    for (int i = 1; i <= n; i++) {
        if (outDeg[i] > 0) {
            return i;
        }
    }
	// start 为单词的首字母
    return -1;
}

void euler(int u, const string& word) {
    for (int e = cur[u]; e != 0; e = cur[u]) {
        cur[u] = nxt[e];
        euler(to[e], weight[e]);
    }

    path[++cntp] = word;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> m;

    for (int i = 1; i <= m; i++) {
        cin >> str[i];

        a[i] = startNode(str[i]);
        b[i] = endNode(str[i]);
        eidArr[i] = i;
    }

    connect();

    int start = directedStart();

    if (start == -1) {
        cout << "***\n";
        return 0;
    }

    euler(start, "");

    // m 条边必须得到 m+1 个顶点，因此 path 中有 m 个有效单词
    if (cntp != m + 1) {
        cout << "***\n";
        return 0;
    }

    cout << path[cntp - 1];

    for (int i = cntp - 2; i >= 1; i--) {
        cout << "." << path[i];
    }

    cout << "\n";
    return 0;
}
```
