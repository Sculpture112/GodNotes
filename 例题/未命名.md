#include <bits/stdc++.h>
using namespace std;

const int BASE = 10000;
typedef vector<int> BigInt;

void trim(BigInt &a) {
    while (a.size() > 1 && a.back() == 0)
        a.pop_back();
}

// 高精度乘小整数
void mul(BigInt &a, int b) {
    long long carry = 0;

    for (int i = 0; i < (int)a.size(); i++) {
        long long cur = 1LL * a[i] * b + carry;
        a[i] = cur % BASE;
        carry = cur / BASE;
    }

    while (carry) {
        a.push_back(carry % BASE);
        carry /= BASE;
    }
}

// 高精度除小整数
BigInt divide(const BigInt &a, int b) {
    BigInt c(a.size());
    long long remainder = 0;

    for (int i = (int)a.size() - 1; i >= 0; i--) {
        long long cur = remainder * BASE + a[i];
        c[i] = cur / b;
        remainder = cur % b;
    }

    trim(c);
    return c;
}

// 判断 a < b
bool lessThan(const BigInt &a, const BigInt &b) {
    if (a.size() != b.size())
        return a.size() < b.size();

    for (int i = (int)a.size() - 1; i >= 0; i--) {
        if (a[i] != b[i])
            return a[i] < b[i];
    }

    return false;
}

void print(const BigInt &a) {
    cout << a.back();

    for (int i = (int)a.size() - 2; i >= 0; i--)
        cout << setw(4) << setfill('0') << a[i];

    cout << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    int kingLeft, kingRight;
    cin >> kingLeft >> kingRight;

    vector<pair<int, int>> minister(n);

    for (int i = 0; i < n; i++)
        cin >> minister[i].first >> minister[i].second;

    sort(minister.begin(), minister.end(),
         [](const pair<int, int> &x, const pair<int, int> &y) {
             return 1LL * x.first * x.second
                  < 1LL * y.first * y.second;
         });

    BigInt product(1, kingLeft);
    BigInt answer(1, 0);

    for (int i = 0; i < n; i++) {
        BigInt reward = divide(product, minister[i].second);

        if (lessThan(answer, reward))
            answer = reward;

        mul(product, minister[i].first);
    }

    print(answer);
    return 0;
}
