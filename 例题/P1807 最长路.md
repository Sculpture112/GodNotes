```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    vector<vector<pair<int, int>>> graph(n + 1);

    for (int i = 0; i < m; ++i) {
        int u, v, w;
        cin >> u >> v >> w;
        graph[u].push_back({v, w});
    }

    const long long NEG_INF = LLONG_MIN / 4;
    vector<long long> dp(n + 1, NEG_INF);
    dp[1] = 0;

    // 题目保证 u < v，因此顶点编号本身就是拓扑序
    for (int u = 1; u <= n; ++u) {
        if (dp[u] == NEG_INF) continue;

        for (auto [v, w] : graph[u]) {
            dp[v] = max(dp[v], dp[u] + w);
        }
    }

    if (dp[n] == NEG_INF)
        cout << -1 << '\n';
    else
        cout << dp[n] << '\n';

    return 0;
}
```

先把支链节点或者说入度为0的节点先计算长度,才能计算主链长度



```cpp
#include <bits/stdc++.h>

// 

using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    vector<vector<pair<int, int>>> graph(n + 1);
    vector<int> indegree(n + 1);

    for (int i = 0; i < m; i++) {
        int u, v, w;
        cin >> u >> v >> w;

        graph[u].push_back({v, w});
        indegree[v]++;
    }

    // Kahn 算法求拓扑序
    queue<int> q;
    for (int i = 1; i <= n; i++) {
        if (indegree[i] == 0) {
            q.push(i);
        }
    }

    vector<int> topo;

    while (!q.empty()) {
        int u = q.front();
        q.pop();
        topo.push_back(u);

        for (auto [v, w] : graph[u]) {
            indegree[v]--;
            if (indegree[v] == 0) {
                q.push(v);
            }
        }
    }

    // 如果拓扑序数量不足，说明存在环
    if ((int)topo.size() != n) {
        cout << -1 << '\n';
        return 0;
    }

    const long long NEG_INF = LLONG_MIN / 4;
    vector<long long> dp(n + 1, NEG_INF);
    dp[1] = 0;

    for (int u : topo) {
        if (dp[u] == NEG_INF) continue;

        for (auto [v, w] : graph[u]) {
            dp[v] = max(dp[v], dp[u] + w);
        }
    }

    cout << (dp[n] == NEG_INF ? -1 : dp[n]) << '\n';

    return 0;
}
```
