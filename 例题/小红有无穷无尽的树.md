#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<vector<int>> graph(n + 1);

    for (int i = 1; i < n; ++i) {
        int u, v;
        cin >> u >> v;
        graph[u].push_back(v);
        graph[v].push_back(u);
    }

    vector<int> parent(n + 1, -1);
    vector<int> depth(n + 1, 0);
    vector<int> childCount(n + 1, 0);
    vector<int> order;
    order.reserve(n);

    vector<int> st;
    st.reserve(n);

    parent[1] = 0;
    st.push_back(1);

    while (!st.empty()) {
        int u = st.back();
        st.pop_back();
        order.push_back(u);

        for (int v : graph[u]) {
            if (v == parent[u]) continue;

            parent[v] = u;
            depth[v] = depth[u] + 1;
            ++childCount[u];
            st.push_back(v);
        }
    }

    vector<int> bestDepth(n + 1, -1);
    vector<int> cnt(n + 1, 0);

    for (int i = n - 1; i >= 0; --i) {
        int u = order[i];

        // u 是叶子，操作后被删除
        if (childCount[u] == 0) {
            bestDepth[u] = -1;
            cnt[u] = 0;
            continue;
        }

        // u 自己不是叶子，会保留下来
        bestDepth[u] = depth[u];
        cnt[u] = 1;

        for (int v : graph[u]) {
            if (parent[v] != u) continue;

            if (bestDepth[v] > bestDepth[u]) {
                bestDepth[u] = bestDepth[v];
                cnt[u] = cnt[v];
            } else if (bestDepth[v] == bestDepth[u]) {
                cnt[u] += cnt[v];
            }
        }
    }

    for (int i = 1; i <= n; ++i) {
        cout << cnt[i] << (i == n ? '\n' : ' ');
    }

    return 0;
}
