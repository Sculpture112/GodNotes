#include <bits/stdc++.h>
using namespace std;

struct Node {
    int x, y; // 无限迷宫中的绝对坐标
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;

    while (cin >> n >> m) {
        vector<string> maze(n);
        int sx = 0, sy = 0;

        for (int i = 0; i < n; ++i) {
            cin >> maze[i];
            for (int j = 0; j < m; ++j) {
                if (maze[i][j] == 'S') {
                    sx = i;
                    sy = j;
                }
            }
        }

        int total = n * m;
        vector<char> visited(total, 0);
        vector<int> originX(total), originY(total);

        auto mod = [](int x, int p) {
            x %= p;
            if (x < 0) x += p;
            return x;
        };

        queue<Node> q;

        int startId = sx * m + sy;
        visited[startId] = 1;
        originX[startId] = sx;
        originY[startId] = sy;
        q.push({sx, sy});

        bool canEscape = false;
        const int dx[4] = {-1, 1, 0, 0};
        const int dy[4] = {0, 0, -1, 1};

        while (!q.empty() && !canEscape) {
            Node cur = q.front();
            q.pop();

            for (int k = 0; k < 4; ++k) {
                int nx = cur.x + dx[k];
                int ny = cur.y + dy[k];

                int rx = mod(nx, n);
                int ry = mod(ny, m);

                if (maze[rx][ry] == '#') continue;

                int id = rx * m + ry;

                if (!visited[id]) {
                    visited[id] = 1;
                    originX[id] = nx;
                    originY[id] = ny;
                    q.push({nx, ny});
                } else if (originX[id] != nx || originY[id] != ny) {
                    canEscape = true;
                    break;
                }
            }
        }

        cout << (canEscape ? "Yes" : "No") << '\n';
    }

    return 0;
}
