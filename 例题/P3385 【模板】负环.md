#include <bits/stdc++.h>
using namespace std;

using ll = long long;

struct Edge {
    int to;
    int weight;
};

bool hasNegativeCycle(int n, const vector<vector<Edge>>& graph) {
    const ll INF = (1LL << 60);

    vector<ll> dist(n + 1, INF);
    vector<int> edgeCount(n + 1, 0);
    vector<bool> inQueue(n + 1, false);
    queue<int> q;

    dist[1] = 0;
    q.push(1);
    inQueue[1] = true; // 判断新加入的点在是否已经在栈里面,提升效率

    while (!q.empty()) {
        int u = q.front();
        q.pop();
        inQueue[u] = false;

        for (const auto& edge : graph[u]) {
            int v = edge.to;
            int w = edge.weight;

            if (dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                edgeCount[v] = edgeCount[u] + 1;

                // 最短路经过至少 n 条边，说明其中存在负环
                if (edgeCount[v] >= n) {
                    return true;
                }

                if (!inQueue[v]) {
                    q.push(v);
                    inQueue[v] = true;
                }
            }
        }
    }

    return false;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int n, m;
        cin >> n >> m;

        vector<vector<Edge>> graph(n + 1);

        for (int i = 0; i < m; ++i) {
            int u, v, w;
            cin >> u >> v >> w;

            graph[u].push_back({v, w});

            // 非负边为双向边
            if (w >= 0) {
                graph[v].push_back({u, w});
            }
        }

        cout << (hasNegativeCycle(n, graph) ? "YES" : "NO") << '\n';
    }

    return 0;
}
