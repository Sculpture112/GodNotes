#include <bits/stdc++.h>
using namespace std;

using int64 = long long;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int n, m;
        cin >> n >> m;

        vector<int64> freq(m + 1, 0);
        vector<int64> suffix(m + 2, 0);

        int64 totalLength = 0;

        for (int i = 0; i < n; ++i) {
            int a;
            cin >> a;

            ++freq[a];
            totalLength += a;
        }

        // suffix[x]：长度 >= x 的胡萝卜数量
        for (int x = m; x >= 1; --x) {
            suffix[x] = suffix[x + 1] + freq[x];
        }

        // P = 2^k
        int P = 2;

        for (int k = 1; k <= m; ++k) {
            int64 answer;

            if (P > m) {
                // 可以全部切成长度为 1
                answer = totalLength;
            } else {
                answer = 0;

                // 官方结论：只需要枚举 x <= m / P
                for (int x = 1; x <= m / P; ++x) {
                    // a_i = P*x 时，可以贡献 P 根而不是 P-1 根
                    int64 current = freq[P * x];

                    // sum(min(a_i / x, P - 1))
                    for (int j = 1; j < P; ++j) {
                        current += suffix[j * x];
                    }

                    answer = max(answer, current);
                }
            }

            cout << answer << (k == m ? '\n' : ' ');

            // P 超过 m 后，后面的答案全部等于 totalLength
            if (P <= m) {
                P *= 2;
            }
        }
    }

    return 0;
}
