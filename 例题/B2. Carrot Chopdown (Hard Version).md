#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int n, m;
        cin >> n >> m;

        vector<int> freq(m + 2, 0);

        for (int i = 0; i < n; ++i) {
            int a;
            cin >> a;
            ++freq[a];
        }

        // suffix[x] = 长度 >= x 的胡萝卜数量
        vector<int> suffix(m + 3, 0);
        for (int x = m; x >= 1; --x) {
            suffix[x] = suffix[x + 1] + freq[x];
        }

        int answer = 0;

        for (int x = 1; x <= m; ++x) {
            int current = suffix[x];

            // 长度恰好为 2x 的胡萝卜可以额外贡献一根
            if (2 * x <= m) {
                current += freq[2 * x];
            }

            answer = max(answer, current);
        }

        cout << answer << '\n';
    }

    return 0;
}
