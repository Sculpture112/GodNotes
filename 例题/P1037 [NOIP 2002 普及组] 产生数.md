```cpp
#include <bits/stdc++.h>
using namespace std;

bool f[10][10];

void print(__int128 x) {
    if (x >= 10) print(x / 10);
    cout << int(x % 10);
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    string n;
    int k;
    cin >> n >> k;

    // 允许不变换，所以每个数字都能变成自己
    for (int i = 0; i < 10; i++) {
        f[i][i] = true;
    }

    while (k--) {
        int x, y;
        cin >> x >> y;
        f[x][y] = true;
    }

    // f[i][j]：数字 i 能否经过任意次变换成为 j
    for (int mid = 0; mid < 10; mid++) {
        for (int i = 0; i < 10; i++) {
            for (int j = 0; j < 10; j++) {
                if (f[i][mid] && f[mid][j]) {
                    f[i][j] = true;
                }
            }
        }
    }

    __int128 ans = 1;

    for (char c : n) {
        int x = c - '0';
        int cnt = 0;

        for (int j = 0; j < 10; j++) {
            if (f[x][j]) cnt++;
        }

        ans *= cnt;
    }

    print(ans);
    cout << '\n';

    return 0;
}
```
