```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<string> chain;
    int ans = 0;

    for (int i = 0; i < n; ++i) {
        string word;
        cin >> word;

        // 弹出所有不是当前单词前缀的单词
        while (!chain.empty() &&
               word.compare(0, chain.back().size(), chain.back()) != 0) {
            chain.pop_back();
        }

        // 当前单词接到剩余词链末尾
        chain.push_back(word);

        ans = max(ans, static_cast<int>(chain.size()));
    }

    cout << ans << '\n';
    return 0;
}
```
