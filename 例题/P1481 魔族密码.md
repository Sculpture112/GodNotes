```cpp
#include <bits/stdc++.h>
using namespace std;

bool isPrefix(const string& prefix, const string& word) {
    if (prefix.size() > word.size()) {
        return false;
    }

    for (int i = 0; i < prefix.size(); ++i) {
        if (prefix[i] != word[i]) {
            return false;
        }
    }

    return true;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<string> words(n);
    for (string& word : words) {
        cin >> word;
    }

    vector<int> dp(n, 1);
    int ans = 1;

    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < i; ++j) {
            if (isPrefix(words[j], words[i])) {
                dp[i] = max(dp[i], dp[j] + 1);
            }
        }

        ans = max(ans, dp[i]);
    }

    cout << ans << '\n';
    return 0;
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<string> chain;
    int ans = 0;

    for (int i = 0; i < n; ++i) {
        string word;
        cin >> word;

        // 弹出所有不是当前单词前缀的单词
        while (!chain.empty() &&
               word.compare(0, chain.back().size(), chain.back()) != 0) {
            chain.pop_back();
        }

        // 当前单词接到剩余词链末尾
        chain.push_back(word);

        ans = max(ans, static_cast<int>(chain.size()));
    }

    cout << ans << '\n';
    return 0;
}
```
