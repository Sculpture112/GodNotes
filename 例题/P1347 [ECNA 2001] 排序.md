```cpp
#include <bits/stdc++.h>
using namespace std;

// 返回值：
// -1：存在矛盾
//  0：无法确定唯一顺序
//  1：已经确定唯一顺序
int topologicalSort(int n, bool graph[26][26], string &order) {
    int indegree[26] = {};

    // 计算当前图的入度
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            if (graph[i][j]) {
                ++indegree[j];
            }
        }
    }

    bool used[26] = {};
    order.clear();
    bool unique = true;

    for (int step = 0; step < n; ++step) { // 
        vector<int> zero;

        // 找出当前所有入度为 0 且未使用的点
        for (int i = 0; i < n; ++i) {
            if (!used[i] && indegree[i] == 0) {
                zero.push_back(i);
            }
        }

        // 没有可选点，说明存在环
        if (zero.empty()) {
            return -1;
        }

        // 同时有多个可选点，说明顺序不唯一
        if (zero.size() > 1) {
            unique = false;
        }

        // 任取一个入度为 0 的点
        int u = zero[0];
        used[u] = true;
        order.push_back(char('A' + u));

        // 删除该点及其出边
        for (int v = 0; v < n; ++v) {
            if (graph[u][v]) {
                --indegree[v];
            }
        }
    }

    return unique ? 1 : 0;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    bool graph[26][26] = {};

    for (int relation = 1; relation <= m; ++relation) {
        string s;
        cin >> s;

        int u = s[0] - 'A';
        int v = s[2] - 'A';

        // 忽略重复关系，避免重复增加入度
        if (!graph[u][v]) { // 太细节了,防止出现两个点却有两个路径
            graph[u][v] = true;
        }

        string order;
        int result = topologicalSort(n, graph, order);

        if (result == -1) {
            cout << "Inconsistency found after "
                 << relation << " relations.\n";
            return 0;
        }

        if (result == 1) {
            cout << "Sorted sequence determined after "
                 << relation << " relations: "
                 << order << ".\n";
            return 0;
        }
    }

    cout << "Sorted sequence cannot be determined.\n";
    return 0;
}
```

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    bool reach[26][26] = {};

    for (int step = 1; step <= m; ++step) { //单次放入数据,然后立刻进行一系列判断
        char a, op, b;
        cin >> a >> op >> b;

        int u = a - 'A';
        int v = b - 'A';
        reach[u][v] = true;

        // Floyd 维护传递闭包,先取中继点,然后枚举起始和末尾
        for (int k = 0; k < n; ++k)
            for (int i = 0; i < n; ++i)
                for (int j = 0; j < n; ++j)
                    reach[i][j] =
                        reach[i][j] || (reach[i][k] && reach[k][j]);

        // 判断是否存在环
        bool inconsistent = false;
        for (int i = 0; i < n; ++i) {
            if (reach[i][i]) {
                inconsistent = true;
                break;
            }
        }

        if (inconsistent) {
            cout << "Inconsistency found after "
                 << step << " relations.\n";
            return 0;
        }

        // 判断是否所有元素都能比较,这是一个数学常识,记住
        bool determined = true;
        for (int i = 0; i < n && determined; ++i) {
            for (int j = i + 1; j < n; ++j) {
                if (!reach[i][j] && !reach[j][i]) {
                    determined = false;
                    break;
                }
            }
        }

        if (determined) {
            string ans(n, ' ');

            // 前驱数量就是元素在最终序列中的位置
            for (int i = 0; i < n; ++i) {
                int predecessors = 0;
                for (int j = 0; j < n; ++j) {
                    if (reach[j][i]) ++predecessors;
                }
                ans[predecessors] = char('A' + i);
            }

            cout << "Sorted sequence determined after "
                 << step << " relations: "
                 << ans << ".\n";
            return 0;
        }
    }

    cout << "Sorted sequence cannot be determined.\n";
    return 0;
}
```
