## 破题切入点 (思维闪念)

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    bool reach[26][26] = {};

    for (int step = 1; step <= m; ++step) {
        char a, op, b;
        cin >> a >> op >> b;

        int u = a - 'A';
        int v = b - 'A';
        reach[u][v] = true;

        // Floyd 维护传递闭包
        for (int k = 0; k < n; ++k)
            for (int i = 0; i < n; ++i)
                for (int j = 0; j < n; ++j)
                    reach[i][j] =
                        reach[i][j] || (reach[i][k] && reach[k][j]);

        // 判断是否存在环
        bool inconsistent = false;
        for (int i = 0; i < n; ++i) {
            if (reach[i][i]) {
                inconsistent = true;
                break;
            }
        }

        if (inconsistent) {
            cout << "Inconsistency found after "
                 << step << " relations.\n";
            return 0;
        }

        // 判断是否所有元素都能比较
        bool determined = true;
        for (int i = 0; i < n && determined; ++i) {
            for (int j = i + 1; j < n; ++j) {
                if (!reach[i][j] && !reach[j][i]) {
                    determined = false;
                    break;
                }
            }
        }

        if (determined) {
            string ans(n, ' ');

            // 前驱数量就是元素在最终序列中的位置
            for (int i = 0; i < n; ++i) {
                int predecessors = 0;
                for (int j = 0; j < n; ++j) {
                    if (reach[j][i]) ++predecessors;
                }
                ans[predecessors] = char('A' + i);
            }

            cout << "Sorted sequence determined after "
                 << step << " relations: "
                 << ans << ".\n";
            return 0;
        }
    }

    cout << "Sorted sequence cannot be determined.\n";
    return 0;
}
```
