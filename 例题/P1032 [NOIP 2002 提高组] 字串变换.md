#include <bits/stdc++.h>
using namespace std;

using Rule = pair<string, string>;

unordered_map<string, int> bfs(
    const string& start,
    const vector<Rule>& rules,
    bool reverseRules
) {
    unordered_map<string, int> dist;
    queue<string> q;

    dist[start] = 0;
    q.push(start);

    while (!q.empty()) {
        string cur = q.front();
        q.pop();

        int step = dist[cur];
        if (step == 5) continue;

        for (auto& rule : rules) {
            string from = reverseRules ? rule.second : rule.first;
            string to   = reverseRules ? rule.first  : rule.second;

            size_t pos = cur.find(from);
            while (pos != string::npos) {
                string nxt = cur.substr(0, pos) +
                             to +
                             cur.substr(pos + from.size());

                if (!dist.count(nxt)) {
                    dist[nxt] = step + 1;
                    q.push(nxt);
                }

                // +1 保证可以处理重叠匹配
                pos = cur.find(from, pos + 1);
            }
        }
    }

    return dist;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    string A, B;
    cin >> A >> B;

    vector<Rule> rules;
    string x, y;
    while (cin >> x >> y) {
        rules.push_back({x, y});
    }

    if (A == B) {
        cout << 0 << '\n';
        return 0;
    }

    // 正向最多搜索 5 步，反向最多搜索 5 步
    auto fromA = bfs(A, rules, false);
    auto fromB = bfs(B, rules, true);

    int ans = INT_MAX;

    for (auto& [s, d1] : fromA) {
        auto it = fromB.find(s);
        if (it != fromB.end()) {
            ans = min(ans, d1 + it->second);
        }
    }

    if (ans <= 10) cout << ans << '\n';
    else cout << "NO ANSWER!\n";

    return 0;
}
