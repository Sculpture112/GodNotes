#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<double> x(n), y(n);
    for (int i = 0; i < n; ++i) {
        cin >> x[i] >> y[i];
    }

    auto distance = [&](int a, int b) {
        double dx = x[a] - x[b];
        double dy = y[a] - y[b];
        return sqrt(dx * dx + dy * dy);
    };

    // distStart[i]：原点到第 i 块奶酪的距离
    vector<double> distStart(n);
    for (int i = 0; i < n; ++i) {
        distStart[i] = sqrt(x[i] * x[i] + y[i] * y[i]);
    }

    // dist[i][j]：第 i 块奶酪到第 j 块奶酪的距离
    vector<vector<double>> dist(n, vector<double>(n));
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            dist[i][j] = distance(i, j);
        }
    }

    int total = 1 << n;
    const double INF = 1e100;

    // dp[mask][i]：吃完 mask 中的奶酪，最后停在 i 的最短距离
    vector<vector<double>> dp(total, vector<double>(n, INF));

    for (int i = 0; i < n; ++i) {
        dp[1 << i][i] = distStart[i];
    }

    for (int mask = 1; mask < total; ++mask) {
        for (int last = 0; last < n; ++last) {
            if (!(mask & (1 << last))) continue;

            for (int nxt = 0; nxt < n; ++nxt) {
                if (mask & (1 << nxt)) continue;

                int newMask = mask | (1 << nxt);
                dp[newMask][nxt] = min(
                    dp[newMask][nxt],
                    dp[mask][last] + dist[last][nxt]
                );
            }
        }
    }

    double answer = INF;
    for (int i = 0; i < n; ++i) {
        answer = min(answer, dp[total - 1][i]);
    }

    cout << fixed << setprecision(2) << answer << '\n';
    return 0;
}
