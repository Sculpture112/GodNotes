#include <bits/stdc++.h>
using namespace std;

int basis[30];

// 成功插入返回 true，否则返回 false
bool insertBasis(int x) {
    for (int bit = 29; bit >= 0; --bit) {
        if (((x >> bit) & 1) == 0) continue;

        if (basis[bit] == 0) {
            basis[bit] = x;
            return true;
        }

        x ^= basis[bit];
    }
    return false;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int k;
    cin >> k;

    vector<int> a(k);
    for (int &x : a) cin >> x;

    sort(a.begin(), a.end(), greater<int>());

    long long ans = 0;
    for (int x : a) {
        if (!insertBasis(x)) {
            ans += x;
        }
    }

    cout << ans << '\n';
    return 0;
}
