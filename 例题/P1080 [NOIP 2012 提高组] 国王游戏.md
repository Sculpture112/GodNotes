#include <bits/stdc++.h>
using namespace std;

using ll = long long;

const int INF = 0x3f3f3f3f;
const ll LINF = 4'000'000'000'000'000'000LL;

#define all(x) (x).begin(), (x).end()

// 高精度整数乘小整数
string multiplySmall(const string &a, int b) {
    string result;
    ll carry = 0;

    // 从个位开始计算
    for (int i = (int)a.size() - 1; i >= 0; --i) {
        ll cur = 1LL * (a[i] - '0') * b + carry;

        result.push_back(char(cur % 10 + '0'));
        carry = cur / 10;
    }

    while (carry > 0) {
        result.push_back(char(carry % 10 + '0'));
        carry /= 10;
    }

    reverse(all(result));
    return result;
}

// 高精度整数除小整数
string divideSmall(const string &a, int b) {
    string result;
    ll remainder = 0;

    // 除法需要从最高位开始
    for (char digit : a) {
        ll cur = remainder * 10 + (digit - '0');

        result.push_back(char(cur / b + '0'));
        remainder = cur % b;
    }

    // 删除商的前导零
    int start = 0;

    while (start + 1 < (int)result.size() &&
           result[start] == '0') {
        ++start;
    }

    return result.substr(start);
}

// 判断高精度整数 a < b
bool lessThan(const string &a, const string &b) {
    if (a.size() != b.size()) {
        return a.size() < b.size();
    }

    return a < b;
}

void solve() {
    int n;
    cin >> n;

    int kingLeft, kingRight;
    cin >> kingLeft >> kingRight;

    vector<pair<int, int>> minister(n);

    for (auto &[left, right] : minister) {
        cin >> left >> right;
    }

    // 按照 左手数 × 右手数 从小到大排列
    sort(all(minister),
         [](const pair<int, int> &x,
            const pair<int, int> &y) {
             return 1LL * x.first * x.second
                  < 1LL * y.first * y.second;
         });

    // 当前排在前面的所有人的左手数乘积
    string product = to_string(kingLeft);

    // 所有大臣中奖赏的最大值
    string answer = "0";

    for (const auto &[left, right] : minister) {
        // 当前大臣获得的金币
        string reward = divideSmall(product, right);

        if (lessThan(answer, reward)) {
            answer = reward;
        }

        // 当前大臣加入前缀乘积，供后面的大臣使用
        product = multiplySmall(product, left);
    }

    cout << answer << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;

    while (T--) solve();

    return 0;
}
