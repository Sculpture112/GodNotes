#include <iostream>
#include <string>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    string n;
    int k;
    cin >> n >> k;

    string ans;

    for (char digit : n) {
        while (k > 0 && !ans.empty() && ans.back() > digit) {
            ans.pop_back();
            --k;
        }
        ans.push_back(digit);
    }

    // 数字整体单调不降时，从末尾删除剩余数字
    while (k > 0) {
        ans.pop_back();
        --k;
    }

    // 去除前导零
    size_t pos = ans.find_first_not_of('0');

    if (pos == string::npos) {
        cout << 0 << '\n';
    } else {
        cout << ans.substr(pos) << '\n';
    }

    return 0;
}
