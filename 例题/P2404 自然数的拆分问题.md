#include <bits/stdc++.h>
using namespace std;

int n;
vector<int> path;

void dfs(int sum, int last) {
    for (int x = last; x < n && sum + x <= n; ++x) {
        path.push_back(x);

        if (sum + x == n) {
            for (int i = 0; i < (int)path.size(); ++i) {
                if (i) cout << '+';
                cout << path[i];
            }
            cout << '\n';
        } else {
            dfs(sum + x, x);
        }

        path.pop_back();
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n;
    dfs(0, 1);

    return 0;
}
