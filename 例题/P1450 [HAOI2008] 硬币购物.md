```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

const int MAXS = 100000;

int c[4];

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;

    for (int i = 0; i < 4; i++) {
        cin >> c[i];
    }

    cin >> n;

    vector<ll> f(MAXS + 1, 0);
    f[0] = 1;

    for (int i = 0; i < 4; i++) {
        for (int j = c[i]; j <= MAXS; j++) {
            f[j] += f[j - c[i]];
        }
    }

    while (n--) {
        int d[4], s;

        for (int i = 0; i < 4; i++) {
            cin >> d[i];
        }

        cin >> s;

        ll ans = 0;

        for (int mask = 0; mask < 16; mask++) {
            int rest = s;
            int cnt = 0;

            for (int i = 0; i < 4; i++) {
                if (mask >> i & 1) {
                    rest -= (d[i] + 1) * c[i];
                    cnt++;
                }
            }

            if (rest < 0) continue;

            if (cnt % 2 == 0) {
                ans += f[rest];
            } else {
                ans -= f[rest];
            }
        }

        cout << ans << '\n';
    }

    return 0;
}
```
