#include <iostream>
#include <vector>
#include <queue>
#include <functional>
#include <limits>
using namespace std;

using ll = long long;
using PII = pair<ll, int>;

struct Edge {
    int to;
    int weight;
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;

    while (cin >> n >> m && (n != 0 || m != 0)) {
        vector<vector<Edge>> graph(n + 1);

        for (int i = 0; i < m; ++i) {
            int a, b, c;
            cin >> a >> b >> c;

            // 道路是双向的
            graph[a].push_back({b, c});
            graph[b].push_back({a, c});
        }

        const ll INF = numeric_limits<ll>::max() / 4;
        vector<ll> dist(n + 1, INF);
        priority_queue<PII, vector<PII>, greater<PII>> heap;

        dist[1] = 0;
        heap.push({0, 1});

        while (!heap.empty()) {
            auto [currentDist, u] = heap.top();
            heap.pop();

            // 跳过已经失效的旧记录
            if (currentDist > dist[u]) {
                continue;
            }

            if (u == n) {
                break;
            }

            for (const Edge& edge : graph[u]) {
                int v = edge.to;
                ll newDist = currentDist + edge.weight;

                if (newDist < dist[v]) {
                    dist[v] = newDist;
                    heap.push({dist[v], v});
                }
            }
        }

        cout << dist[n] << '\n';
    }

    return 0;
}
