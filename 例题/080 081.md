 每次从剩余任务里挑一个“能塞进一个时间段的子集 sub”，作为一个完整时间段。
 
 class Solution {
public:
    int minSessions(vector<int>& tasks, int sessionTime) {
        int n = tasks.size();
        int size = 1 << n;

        vector<int> cost(size, 0);
        for (int s = 1; s < size; s++) {
            int rightOne = s & -s;
            int i = __builtin_ctz(rightOne);
            cost[s] = cost[s ^ rightOne] + tasks[i];
        }

        const int INF = 1e9;
        vector<int> dp(size, INF);
        dp[0] = 0;

        for (int status = 1; status < size; status++) {
            for (int sub = status; sub > 0; sub = (sub - 1) & status) {
            //这里遍历二进制
                if (cost[sub] <= sessionTime) {
                    dp[status] = min(dp[status], dp[status ^ sub] + 1);
                }
            }
        }

        return dp[size - 1];
    }
};

class Solution {
public:
    int minSessions(vector<int>& tasks, int sessionTime) {
        int n = tasks.size();
        int size = 1 << n;

        vector<pair<int, int>> dp(size, {n + 1, 0});
        dp[0] = {1, 0};

        for (int status = 1; status < size; status++) {
            for (int i = 0; i < n; i++) {
                if ((status & (1 << i)) != 0) {
                    auto pre = dp[status ^ (1 << i)];

                    if (pre.second + tasks[i] <= sessionTime) {
                        pre.second += tasks[i];
                    } else {
                        pre.first++;
                        pre.second = tasks[i];
                    }

                    dp[status] = min(dp[status], pre);
                }
            }
        }

        return dp[size - 1].first;
    }
};
