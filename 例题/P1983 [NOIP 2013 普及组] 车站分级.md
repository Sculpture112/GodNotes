## 破题切入点 (思维闪念)

```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 1005;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    // adj[u][v] = 1 表示存在边 u -> v
    vector<bitset<MAXN>> adj(n + 1);

    for (int i = 0; i < m; ++i) {
        int s;
        cin >> s;

        vector<int> stops(s);
        vector<bool> isStop(n + 1, false);
        bitset<MAXN> stopBits;

        for (int j = 0; j < s; ++j) {
            cin >> stops[j];
            isStop[stops[j]] = true;
            stopBits.set(stops[j]);
        }

        int left = stops.front();
        int right = stops.back();

        // 区间内未停靠的车站，必须低于所有停靠站
        for (int station = left; station <= right; ++station) {
            if (!isStop[station]) {
                adj[station] |= stopBits;
            }
        }
    }

    vector<int> indegree(n + 1, 0);

    // 统计入度
    for (int u = 1; u <= n; ++u) {
        for (int v = 1; v <= n; ++v) {
            if (adj[u][v]) {
                ++indegree[v];
            }
        }
    }

    queue<int> q;
    vector<int> dp(n + 1, 1);

    for (int i = 1; i <= n; ++i) {
        if (indegree[i] == 0) {
            q.push(i);
        }
    }

    int answer = 1;

    // 拓扑排序 + 最长路
    while (!q.empty()) {
        int u = q.front();
        q.pop();

        answer = max(answer, dp[u]); // 拓扑排序,还可以弹出的时候取最大值

        for (int v = 1; v <= n; ++v) {
            if (adj[u][v]) {
                dp[v] = max(dp[v], dp[u] + 1);
                --indegree[v];

                if (indegree[v] == 0) {
                    q.push(v);
                }
            }
        }
    }

    cout << answer << '\n';
    return 0;
}
```
