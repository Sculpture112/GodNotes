#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n;
    cin >> n;

    vector<string> a(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }

    string word = "yizhong";
    vector<vector<bool>> marked(n, vector<bool>(n, false));

    int dx[8] = {-1, -1, -1, 0, 0, 1, 1, 1};
    int dy[8] = {-1, 0, 1, -1, 1, -1, 0, 1};

    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            for (int d = 0; d < 8; ++d) {
                int x = i, y = j;
                bool ok = true;

                for (int k = 0; k < 7; ++k) {
                    if (x < 0 || x >= n || y < 0 || y >= n ||
                        a[x][y] != word[k]) {
                        ok = false;
                        break;
                    }
                    x += dx[d];
                    y += dy[d];
                }

                if (ok) {
                    x = i;
                    y = j;
                    for (int k = 0; k < 7; ++k) {
                        marked[x][y] = true;
                        x += dx[d];
                        y += dy[d];
                    }
                }
            }
        }
    }

    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            cout << (marked[i][j] ? a[i][j] : '*');
        }
        cout << '\n';
    }

    return 0;
}
