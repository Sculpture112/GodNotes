#include <bits/stdc++.h>
using namespace std;

int n;
char ch;
string s[25];
int used[25];
int addLen[25][25];
int ans = 0;

int getOverlap(const string& a, const string& b) {
    int limit = min(a.size(), b.size()) - 1;

    // 取最小合法重叠长度
    for (int len = 1; len <= limit; ++len) {
        bool ok = true;

        for (int k = 0; k < len; ++k) {
            if (a[a.size() - len + k] != b[k]) {
                ok = false;
                break;
            }
        }

        if (ok) return len;
    }

    return 0;
}

void dfs(int last, int length) {
    ans = max(ans, length);

    for (int i = 0; i < n; ++i) {
        if (used[i] >= 2 || addLen[last][i] == 0)
            continue;

        ++used[i];
        dfs(i, length + addLen[last][i]);
        --used[i];
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n;

    for (int i = 0; i < n; ++i)
        cin >> s[i];

    cin >> ch;

    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            int overlap = getOverlap(s[i], s[j]);
            if (overlap)
                addLen[i][j] = s[j].size() - overlap;
        }
    }

    for (int i = 0; i < n; ++i) {
        if (s[i][0] == ch) {
            used[i] = 1;
            dfs(i, s[i].size());
            used[i] = 0;
        }
    }

    cout << ans << '\n';
    return 0;
}
